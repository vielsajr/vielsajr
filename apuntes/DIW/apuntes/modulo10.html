// formulario_ajax.js — envío sin recarga con validación
document.getElementById('form-registro')
  .addEventListener('submit', async (event) => {
    event.preventDefault(); // ← FUNDAMENTAL: evitar recarga de página

    const form   = event.target;
    const btn    = document.getElementById('btn-submit');
    const msgDiv = document.getElementById('mensajes');

    msgDiv.innerHTML = '';
    btn.disabled = true;
    btn.querySelector('.btn-texto').textContent = 'Enviando...';

    // Recoger datos: FormData + Object.fromEntries → objeto JS
    const datos = Object.fromEntries(new FormData(form));

    try {
      const response = await fetch('/api/registro', {
        method:  'POST',
        headers: { 'Content-Type': 'application/json' },
        body:    JSON.stringify(datos),
      });
      const resultado = await response.json();

      if (!response.ok) {
        if (response.status === 422 && resultado.errores) {
          // Mostrar errores de validación campo a campo
          const html = Object.entries(resultado.errores)
            .map(([campo, errores]) =>
              `<p class="error">${campo}: ${errores.join(', ')}</p>`
            ).join('');
          msgDiv.innerHTML = html;
        } else {
          throw new Error(resultado.mensaje || `Error ${response.status}`);
        }
        return;
      }

      // Éxito
      form.reset();
      msgDiv.innerHTML = '<p class="success">¡Registro completado!</p>';
      setTimeout(() => window.location.href = '/dashboard', 1500);

    } catch (error) {
      msgDiv.innerHTML = `<p class="error">${error.message}</p>`;
    } finally {
      btn.disabled = false;
      btn.querySelector('.btn-texto').textContent = 'Registrarse';
    }
  });

10.2 Búsqueda en tiempo real (Live Search)

live_search.js
// live_search.js — debounce + AbortController

// Función debounce: retrasa la ejecución hasta que paren de escribir
function debounce(fn, delay) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

const resultadosDiv = document.getElementById('resultados-busqueda');
let abortCtrl = null;

async function buscar(termino) {
  if (termino.length < 2) { resultadosDiv.innerHTML = ''; return; }

  if (abortCtrl) abortCtrl.abort(); // cancelar petición anterior
  abortCtrl = new AbortController();
  resultadosDiv.innerHTML = '<p>Buscando...</p>';

  try {
    const params = new URLSearchParams({ q: termino, limit: 10 });
    const res = await fetch(`/api/productos?${params}`, {
      signal: abortCtrl.signal
    });
    const productos = await res.json();

    if (productos.length === 0) {
      resultadosDiv.innerHTML = '<p>Sin resultados</p>';
      return;
    }

    const html = productos.map(p => `
      <div class="resultado-item" data-id="${p.id}">
        <strong>${resaltarTermino(p.nombre, termino)}</strong>
        <span>${p.precio.toFixed(2)} €</span>
      </div>
    `).join('');
    resultadosDiv.innerHTML = html;

    resultadosDiv.querySelectorAll('.resultado-item').forEach(el => {
      el.addEventListener('click', () => {
        window.location.href = `/productos/${el.dataset.id}`;
      });
    });

  } catch (err) {
    if (err.name !== 'AbortError') {
      resultadosDiv.innerHTML = '<p class="error">Error al buscar</p>';
    }
  }
}

function resaltarTermino(texto, termino) {
  const regex = new RegExp(`(${termino})`, 'gi');
  return texto.replace(regex, '<mark>$1</mark>');
}

const buscarDebounced = debounce(buscar, 300);
document.getElementById('buscador')
  .addEventListener('input', e => buscarDebounced(e.target.value.trim()));

10.3 Paginación infinita con IntersectionObserver

paginacion_infinita.js
// paginacion_infinita.js — cargar más al llegar al final con IntersectionObserver
let pagina   = 1;
let cargando = false;
let hayMas   = true;

const lista      = document.getElementById('lista-items');
const spinnerFin = document.getElementById('spinner-final');

async function cargarPagina() {
  if (cargando || !hayMas) return;
  cargando = true;
  spinnerFin.style.display = 'block';

  try {
    const res  = await fetch(`/api/items?page=${pagina}&per_page=20`);
    const datos = await res.json();

    datos.items.forEach(item => {
      const el = document.createElement('div');
      el.className = 'item-card';
      el.innerHTML = `
        <h3>${item.titulo}</h3>
        <p>${item.descripcion}</p>
        <span class="precio">${item.precio} €</span>
      `;
      lista.appendChild(el);
    });

    hayMas = datos.pagina_actual < datos.total_paginas;
    pagina++;
    if (!hayMas) spinnerFin.textContent = 'No hay más items';

  } catch (err) {
    console.error('Error cargando página:', err.message);
  } finally {
    cargando = false;
    if (hayMas) spinnerFin.style.display = 'none';
  }
}

// IntersectionObserver — detectar cuando el spinner entra en el viewport
const observer = new IntersectionObserver((entries) => {
  if (entries[0].isIntersecting) cargarPagina();
}, { threshold: 0.1 });

observer.observe(spinnerFin);
cargarPagina(); // cargar primera página

// crud_dinamico.js — gestión de tareas con AJAX en tiempo real
const lista = document.getElementById('lista-tareas');

function crearElementoTarea(tarea) {
  const li = document.createElement('li');
  li.dataset.id = tarea.id;
  li.className  = `tarea ${tarea.completada ? 'completada' : ''}`;
  li.innerHTML  = `
    <input type="checkbox" class="check" ${tarea.completada ? 'checked' : ''}>
    <span class="texto">${tarea.titulo}</span>
    <button class="btn-editar">Editar</button>
    <button class="btn-eliminar">Eliminar</button>
  `;

  // Marcar/desmarcar
  li.querySelector('.check').addEventListener('change', async (e) => {
    try {
      await fetch(`/api/tareas/${tarea.id}`, {
        method:  'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body:    JSON.stringify({ completada: e.target.checked }),
      });
      li.classList.toggle('completada', e.target.checked);
    } catch (err) {
      e.target.checked = !e.target.checked; // revertir si falla
      alert('Error al actualizar');
    }
  });

  // Eliminar
  li.querySelector('.btn-eliminar').addEventListener('click', async () => {
    if (!confirm(`¿Eliminar '${tarea.titulo}'?`)) return;
    try {
      await fetch(`/api/tareas/${tarea.id}`, { method: 'DELETE' });
      li.style.animation = 'fadeOut 0.3s ease';
      setTimeout(() => li.remove(), 300);
    } catch (err) {
      alert('Error al eliminar');
    }
  });

  return li;
}

async function cargarTareas() {
  lista.innerHTML = '<li>Cargando...</li>';
  try {
    const tareas = await fetch('/api/tareas').then(r => r.json());
    lista.innerHTML = '';
    tareas.forEach(t => lista.appendChild(crearElementoTarea(t)));
  } catch (err) {
    lista.innerHTML = '<li class="error">Error al cargar</li>';
  }
}

document.getElementById('form-nueva-tarea').addEventListener('submit', async (e) => {
  e.preventDefault();
  const input  = e.target.querySelector('input[name=titulo]');
  const titulo = input.value.trim();
  if (!titulo) return;
  try {
    const res   = await fetch('/api/tareas', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ titulo }),
    });
    const nueva = await res.json();
    lista.prepend(crearElementoTarea(nueva));
    input.value = '';
  } catch (err) {
    alert('Error al crear la tarea');
  }
});

cargarTareas();
