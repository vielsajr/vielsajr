14.2 Ejercicio 1 — Fetch básico con manejo de errores

📝 EJERCICIO 1 — Nivel básico

Función obtenerProducto(id) con Fetch

Escribe una función obtenerProducto(id) que use Fetch para obtener un producto de /api/productos/{id}. Debe comprobar response.ok, devolver el objeto JSON si hay éxito, y lanzar un Error con el código HTTP si falla. Llama a la función y muestra el nombre en un div con id="resultado".

Solución Ejercicio 1
async function obtenerProducto(id) {
  const response = await fetch(`/api/productos/${id}`);
  if (!response.ok) {
    throw new Error(`Error ${response.status}: ${response.statusText}`);
  }
  return response.json();
}

const resultado = document.getElementById('resultado');
obtenerProducto(42)
  .then(producto => {
    resultado.textContent = producto.nombre;
    resultado.className   = 'exito';
  })
  .catch(error => {
    resultado.textContent = `Error: ${error.message}`;
    resultado.className   = 'error';
  });

14.3 Ejercicio 2 — Formulario AJAX con spinner

📝 EJERCICIO 2 — Nivel medio

Formulario de login con AJAX

Crea el JS para que el formulario con id="form-login" envíe datos por AJAX (POST a /api/login con JSON). Debe: evitar la recarga, mostrar un spinner mientras carga, mostrar el error si status 401, redirigir a /dashboard si status 200, y deshabilitar el botón durante la petición.

Solución Ejercicio 2
document.getElementById('form-login').addEventListener('submit', async (e) => {
  e.preventDefault();
  const form    = e.target;
  const btn     = form.querySelector('[type=submit]');
  const msgDiv  = document.getElementById('mensaje-error');
  const spinner = document.getElementById('spinner');

  btn.disabled         = true;
  spinner.style.display = 'block';
  msgDiv.textContent   = '';

  const datos = Object.fromEntries(new FormData(form));

  try {
    const res  = await fetch('/api/login', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify(datos),
    });
    const json = await res.json();

    if (res.status === 401) {
      msgDiv.textContent = json.mensaje || 'Credenciales incorrectas';
      return;
    }
    if (!res.ok) throw new Error(json.mensaje || `Error ${res.status}`);

    localStorage.setItem('token', json.token);
    window.location.href = '/dashboard';

  } catch (error) {
    msgDiv.textContent = error.message;
  } finally {
    btn.disabled          = false;
    spinner.style.display = 'none';
  }
});

14.4 Ejercicio 3 — Búsqueda con debounce y AbortController

📝 EJERCICIO 3 — Nivel avanzado

Live search con debounce 400ms

Implementa una búsqueda en tiempo real en el input con id="search". Requisitos: no hacer petición si el input tiene menos de 3 caracteres, usar debounce de 400ms, cancelar la petición anterior si el usuario sigue escribiendo, mostrar resultados en div id="results" con nombre y precio, manejar AbortError sin mostrarlo al usuario.

Solución Ejercicio 3
const searchInput = document.getElementById('search');
const resultsDiv  = document.getElementById('results');
let currentCtrl   = null;

function debounce(fn, delay) {
  let timer;
  return (...args) => { clearTimeout(timer); timer = setTimeout(() => fn(...args), delay); };
}

async function buscar(termino) {
  if (termino.length < 3) { resultsDiv.innerHTML = ''; return; }

  if (currentCtrl) currentCtrl.abort();
  currentCtrl = new AbortController();

  try {
    const params = new URLSearchParams({ q: termino });
    const res    = await fetch(`/api/productos?${params}`, {
      signal: currentCtrl.signal
    });
    const items = await res.json();

    resultsDiv.innerHTML = items.length === 0
      ? '<p>Sin resultados</p>'
      : items.map(i =>
          `<div class="result"><b>${i.nombre}</b> — ${i.precio} €</div>`
        ).join('');

  } catch (err) {
    if (err.name !== 'AbortError') {
      resultsDiv.innerHTML = '<p class="error">Error al buscar</p>';
    }
  }
}

searchInput.addEventListener('input', debounce(e => buscar(e.target.value.trim()), 400));
