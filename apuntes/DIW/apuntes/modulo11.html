11.1 Endpoint PHP completo para AJAX

El servidor PHP recibe las peticiones AJAX igual que cualquier petición HTTP. La diferencia es que la respuesta debe ser JSON (no HTML), y debemos configurar las cabeceras CORS correctamente. Se usa file_get_contents("php://input") para leer el body JSON.
api/usuarios.php
<?php
// api/usuarios.php — endpoint PHP para AJAX

// ── 1. Cabeceras CORS y Content-Type ─────────────────────────────────
header('Content-Type: application/json; charset=UTF-8');
header('Access-Control-Allow-Origin: *');
header('Access-Control-Allow-Methods: GET, POST, PUT, PATCH, DELETE, OPTIONS');
header('Access-Control-Allow-Headers: Content-Type, Authorization');

if ($_SERVER['REQUEST_METHOD'] === 'OPTIONS') {
  http_response_code(204); exit;
}

// ── 2. Helpers de respuesta ───────────────────────────────────────────
function jsonResponse($data, $statusCode = 200): void {
  http_response_code($statusCode);
  echo json_encode($data, JSON_UNESCAPED_UNICODE | JSON_PRETTY_PRINT);
  exit;
}
function jsonError($mensaje, $statusCode = 400, $errores = []): void {
  $body = ['error' => true, 'mensaje' => $mensaje];
  if (!empty($errores)) $body['errores'] = $errores;
  jsonResponse($body, $statusCode);
}

// ── 3. Leer cuerpo JSON de la petición ───────────────────────────────
function getBodyJSON(): array {
  $raw = file_get_contents('php://input');
  if (empty($raw)) return [];
  $data = json_decode($raw, true);
  if (json_last_error() !== JSON_ERROR_NONE) jsonError('JSON malformado', 400);
  return $data ?? [];
}

// ── 4. Enrutamiento por método ────────────────────────────────────────
$method = $_SERVER['REQUEST_METHOD'];
$id     = isset($_GET['id']) ? (int)$_GET['id'] : null;

require_once '../includes/db.php';
$pdo = getDB();

switch ($method) {
  case 'GET':
    if ($id) {
      $stmt = $pdo->prepare('SELECT id, nombre, email FROM usuarios WHERE id = ?');
      $stmt->execute([$id]);
      $usuario = $stmt->fetch();
      if (!$usuario) jsonError('Usuario no encontrado', 404);
      jsonResponse($usuario);
    } else {
      $page   = max(1, (int)($_GET['page'] ?? 1));
      $limit  = min(100, max(1, (int)($_GET['per_page'] ?? 20)));
      $offset = ($page - 1) * $limit;
      $stmt   = $pdo->prepare('SELECT id, nombre, email FROM usuarios LIMIT ? OFFSET ?');
      $stmt->execute([$limit, $offset]);
      $total  = $pdo->query('SELECT COUNT(*) FROM usuarios')->fetchColumn();
      jsonResponse([
        'datos'        => $stmt->fetchAll(),
        'total'        => (int)$total,
        'pagina'       => $page,
        'por_pagina'   => $limit,
        'total_paginas'=> (int)ceil($total / $limit),
      ]);
    }
    break;

  case 'POST':
    $datos   = getBodyJSON();
    $errores = validarUsuario($datos);
    if (!empty($errores)) jsonError('Validación fallida', 422, $errores);
    $stmt = $pdo->prepare('INSERT INTO usuarios (nombre, email, password) VALUES (?,?,?)');
    $stmt->execute([
      htmlspecialchars(trim($datos['nombre'])),
      filter_var($datos['email'], FILTER_SANITIZE_EMAIL),
      password_hash($datos['password'], PASSWORD_BCRYPT),
    ]);
    $nuevo = $pdo->query('SELECT id, nombre, email FROM usuarios WHERE id = LAST_INSERT_ID()')->fetch();
    jsonResponse($nuevo, 201);
    break;

  case 'PATCH':
    if (!$id) jsonError('Se requiere ID', 400);
    $datos = getBodyJSON();
    $stmt  = $pdo->prepare('UPDATE usuarios SET nombre = COALESCE(?, nombre), email = COALESCE(?, email) WHERE id = ?');
    $stmt->execute([$datos['nombre'] ?? null, $datos['email'] ?? null, $id]);
    if ($stmt->rowCount() === 0) jsonError('Usuario no encontrado', 404);
    jsonResponse(['mensaje' => 'Actualizado correctamente']);
    break;

  case 'DELETE':
    if (!$id) jsonError('Se requiere ID', 400);
    $stmt = $pdo->prepare('DELETE FROM usuarios WHERE id = ?');
    $stmt->execute([$id]);
    if ($stmt->rowCount() === 0) jsonError('Usuario no encontrado', 404);
    jsonResponse(null, 204);
    break;

  default: jsonError('Método no permitido', 405);
}

function validarUsuario(array $datos): array {
  $errores = [];
  if (empty($datos['nombre']))
    $errores['nombre'][] = 'El nombre es obligatorio';
  if (empty($datos['email']) || !filter_var($datos['email'], FILTER_VALIDATE_EMAIL))
    $errores['email'][]  = 'Email inválido';
  if (!empty($datos['password']) && strlen($datos['password']) < 8)
    $errores['password'][] = 'Mínimo 8 caracteres';
  return $errores;
}

11.2 Autenticación con JWT en AJAX

auth_ajax.js
// auth_ajax.js — login y gestión de token JWT

// ── Login: obtener token ───────────────────────────────────────────────
async function login(email, password) {
  const res = await fetch('/api/auth/login', {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify({ email, password }),
  });
  if (!res.ok) {
    const err = await res.json();
    throw new Error(err.mensaje || 'Credenciales incorrectas');
  }
  const { token, usuario } = await res.json();

  // Guardar token (sessionStorage = sesión, localStorage = persistente)
  localStorage.setItem('token',   token);
  localStorage.setItem('usuario', JSON.stringify(usuario));
  return usuario;
}

// ── Función wrapper que añade el token automáticamente ────────────────
async function apiFetch(url, opciones = {}) {
  const token = localStorage.getItem('token');
  const headers = {
    'Content-Type': 'application/json',
    ...opciones.headers,
  };
  if (token) headers['Authorization'] = `Bearer ${token}`;

  const res = await fetch(url, { ...opciones, headers });

  // Token expirado → redirigir a login
  if (res.status === 401) {
    localStorage.removeItem('token');
    localStorage.removeItem('usuario');
    window.location.href = '/login?expired=1';
    return;
  }
  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err.mensaje || `Error ${res.status}`);
  }
  if (res.status === 204) return null;
  return res.json();
}

// ── Logout ────────────────────────────────────────────────────────────
async function logout() {
  try {
    await apiFetch('/api/auth/logout', { method: 'POST' });
  } finally {
    localStorage.removeItem('token');
    localStorage.removeItem('usuario');
    window.location.href = '/login';
  }
}
