5.1 Sintaxis básica async/await

async/await es azúcar sintáctica sobre Promises. Hace que el código asíncrono parezca síncrono. Cualquier función async devuelve siempre una Promise. await solo puede usarse dentro de funciones async.
async_await_basico.js
// ── La misma operación: con .then() vs async/await ───────────────────

// CON .then() (encadenado):
function obtenerDatosConThen() {
  return fetch('/api/usuarios/1')
    .then(r => { if (!r.ok) throw new Error(r.status); return r.json(); })
    .then(usuario => fetch(`/api/pedidos?userId=${usuario.id}`))
    .then(r => r.json());
}

// CON async/await (mucho más claro):
async function obtenerDatosConAwait() {
  // await 'pausa' la función hasta que la Promise se resuelve
  // El hilo de ejecución NO SE BLOQUEA — JS sigue con otras tareas
  const resUsuario = await fetch('/api/usuarios/1');
  if (!resUsuario.ok) throw new Error(`Error ${resUsuario.status}`);
  const usuario = await resUsuario.json();

  const resPedidos = await fetch(`/api/pedidos?userId=${usuario.id}`);
  const pedidos    = await resPedidos.json();

  return { usuario, pedidos }; // Promise automáticamente resuelta
}

// LLAMAR a una función async:
obtenerDatosConAwait()
  .then(({ usuario, pedidos }) => {
    console.log('Usuario:', usuario.name);
    console.log('Pedidos:', pedidos.length);
  })
  .catch(err => console.error(err.message));

// O con await (dentro de otra función async o top-level module):
const { usuario, pedidos } = await obtenerDatosConAwait();

5.2 Manejo de errores con try/catch

async_errores.js
// ── try/catch con async/await ────────────────────────────────────────
async function obtenerUsuario(id) {
  try {
    const response = await fetch(`/api/usuarios/${id}`);
    if (!response.ok) {
      const errorData = await response.json().catch(() => ({}));
      throw new Error(errorData.mensaje || `Error HTTP ${response.status}`);
    }
    const usuario = await response.json();
    return usuario;

  } catch (error) {
    if (error.name === 'AbortError') {
      console.log('Petición cancelada');
      return null;
    }
    if (error.name === 'TypeError') {
      console.error('Error de red:', error.message);
    } else {
      console.error('Error:', error.message);
    }
    throw error; // re-lanzar si el llamador necesita saberlo

  } finally {
    ocultarCargando(); // siempre se ejecuta
  }
}

// ── Patrón: manejo al estilo Go/Node [error, datos] ──────────────────
async function safeAwait(promise) {
  try {
    const data = await promise;
    return [null, data];
  } catch (error) {
    return [error, null];
  }
}

// Uso más limpio (sin try/catch en el llamador):
const [error, usuario] = await safeAwait(obtenerUsuario(1));
if (error) {
  console.error('No se pudo obtener el usuario:', error.message);
} else {
  console.log('Usuario:', usuario.nombre);
}

5.3 Patrones avanzados: paralelo vs secuencial

async_patrones.js
// ── Peticiones en PARALELO (más eficiente) ───────────────────────────

// ❌ MALO: secuencial — espera a que acabe cada una (total: ~3s)
async function cargarSecuencial() {
  const usuarios  = await fetch('/api/usuarios').then(r => r.json()); // ~1s
  const productos = await fetch('/api/productos').then(r => r.json()); // ~1s
  const pedidos   = await fetch('/api/pedidos').then(r => r.json());  // ~1s
  return { usuarios, productos, pedidos };
}

// ✅ BIEN: paralelo — todas a la vez (total: ~1s, el más lento)
async function cargarParalelo() {
  const [usuarios, productos, pedidos] = await Promise.all([
    fetch('/api/usuarios').then(r => r.json()),
    fetch('/api/productos').then(r => r.json()),
    fetch('/api/pedidos').then(r => r.json()),
  ]);
  return { usuarios, productos, pedidos };
}

// ── Flujo que mezcla secuencial (depende de anterior) + paralelo ─────
async function flujoCompleto(emailUsuario) {
  // 1. Login (secuencial — necesita el token para lo siguiente)
  const auth = await fetch('/api/login', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: emailUsuario, pass: '1234' }),
  }).then(r => r.json());

  // 2. Obtener perfil con el token
  const perfil = await fetch('/api/perfil', {
    headers: { Authorization: `Bearer ${auth.token}` }
  }).then(r => r.json());

  // 3. Cargar pedidos y favoritos en PARALELO (no dependen el uno del otro)
  const [pedidos, favoritos] = await Promise.all([
    fetch(`/api/pedidos?userId=${perfil.id}`).then(r => r.json()),
    fetch(`/api/favoritos?userId=${perfil.id}`).then(r => r.json()),
  ]);

  return { perfil, pedidos, favoritos };
}

// ── for...of con await (bucle asíncrono secuencial) ───────────────────
async function procesarIds(ids) {
  const resultados = [];
  for (const id of ids) {
    const datos = await fetch(`/api/items/${id}`).then(r => r.json());
    resultados.push(datos);
    // Útil para throttling o cuando cada petición depende de la anterior
  }
  return resultados;
}
