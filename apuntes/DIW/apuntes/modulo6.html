6.1 Formato JSON y tipos de datos

JSON (JavaScript Object Notation) es el formato estándar para intercambiar datos entre cliente y servidor en aplicaciones AJAX modernas. Es texto plano, ligero y legible por humanos y máquinas.
ejemplo.json
{
  "id": 1,
  "nombre": "Ana García",
  "email": "ana@mail.com",
  "activo": true,
  "edad": 28,
  "saldo": 1234.56,
  "telefono": null,
  "roles": ["usuario", "editor"],
  "direccion": {
    "calle": "Mayor 5",
    "ciudad": "Madrid",
    "cp": "28001"
  },
  "pedidos": [
    { "id": 101, "total": 49.99, "estado": "enviado" },
    { "id": 102, "total": 89.00, "estado": "pendiente" }
  ]
}

// ── Diferencias importantes con objetos JS ────────────────────────────
// JSON: claves SIEMPRE con comillas dobles
// JSON: no admite funciones, undefined, Date, RegExp
// JSON: no admite comentarios
// JSON: no admite comas finales (trailing commas)

6.2 JSON.stringify() y JSON.parse()

json_metodos.js
// ════════════════════════════════════════════════════════════
// JSON.stringify() — Objeto JS → String JSON (para enviar al servidor)
// ════════════════════════════════════════════════════════════
const usuario = {
  id: 1,
  nombre: 'Ana',
  email:  'ana@mail.com',
  creado: new Date(),      // ← se convierte a string ISO 8601
  metodo: function() {},   // ← se OMITE (no existe en JSON)
  indefinido: undefined,   // ← se OMITE
};

// Básico:
const json = JSON.stringify(usuario);
// '{"id":1,"nombre":"Ana","email":"ana@mail.com","creado":"..."}'

// Con formato (útil para depuración):
const jsonBonito = JSON.stringify(usuario, null, 4);

// Con replacer — filtrar propiedades:
const jsonFiltrado = JSON.stringify(usuario, ['id', 'nombre'], 2);

// Con función replacer — transformar:
JSON.stringify(usuario, (clave, valor) => {
  if (clave === 'email') return undefined;    // excluir
  if (typeof valor === 'number') return valor * 2; // transformar
  return valor;
});

// ════════════════════════════════════════════════════════════
// JSON.parse() — String JSON → Objeto JS (para usar los datos)
// ════════════════════════════════════════════════════════════
const jsonString = '{"id":1,"nombre":"Ana","activo":true}';

const obj = JSON.parse(jsonString);
console.log(obj.nombre); // 'Ana'
console.log(obj.activo); // true (boolean, no string)

// Con reviver — transformar valores al parsear:
const objConFechas = JSON.parse(jsonString, (clave, valor) => {
  if (typeof valor === 'string' && /^\d{4}-\d{2}-\d{2}/.test(valor)) {
    return new Date(valor); // convertir fechas automáticamente
  }
  return valor;
});

// ⚠️ SIEMPRE usar try/catch al parsear JSON externo:
function parsearJSON(texto) {
  try {
    return JSON.parse(texto);
  } catch (e) {
    console.error('JSON inválido:', e.message);
    return null;
  }
}
