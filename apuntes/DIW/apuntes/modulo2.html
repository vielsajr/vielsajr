2.1 Estados de XHR (readyState)

XHR fue la primera tecnología que hizo posible AJAX (introducida por Microsoft en 1999). Aunque hoy la Fetch API es preferible, XHR sigue siendo importante en código heredado y en algunos exámenes.
EstadoDescripción
0 — UNSENTEl objeto XHR fue creado pero open() aún no fue llamado.
1 — OPENEDopen() fue llamado. La petición está configurada.
2 — HEADERS_RECEIVEDsend() fue llamado. Se recibieron las cabeceras y el estado HTTP.
3 — LOADINGEl cuerpo de la respuesta se está descargando (responseText tiene datos parciales).
4 — DONELa operación está completa. La respuesta está disponible completamente.

2.2 XHR completo — GET con todos los eventos

xhr_completo.js
const xhr = new XMLHttpRequest();

// open(método, url, asíncrono=true, usuario, contraseña)
xhr.open('GET', 'https://jsonplaceholder.typicode.com/users/1', true);

// ── Cabeceras de petición ─────────────────────────────────────────────
xhr.setRequestHeader('Accept', 'application/json');
xhr.setRequestHeader('Authorization', 'Bearer mi-token-aqui');

// ── Configurar tipo de respuesta ──────────────────────────────────────
xhr.responseType = 'json'; // 'text' | 'json' | 'blob' | 'arraybuffer' | 'document'

// ── Timeout ───────────────────────────────────────────────────────────
xhr.timeout = 5000; // 5 segundos

// ── Evento principal: onreadystatechange ──────────────────────────────
xhr.onreadystatechange = function() {
  console.log('readyState:', xhr.readyState);
  if (xhr.readyState === XMLHttpRequest.DONE) { // === 4
    if (xhr.status >= 200 && xhr.status < 300) {
      const usuario = xhr.response; // ya está parseado con responseType='json'
      console.log('Usuario:', usuario.name);
    } else {
      console.error('Error HTTP:', xhr.status, xhr.statusText);
    }
  }
};

// ── Eventos alternativos (más específicos) ────────────────────────────
xhr.onload = function() {
  if (this.status === 200) console.log(this.response);
};
xhr.onerror   = () => console.error('Error de red o CORS');
xhr.ontimeout = () => console.error('La petición tardó demasiado');
xhr.onprogress = function(event) {
  if (event.lengthComputable) {
    const pct = (event.loaded / event.total) * 100;
    console.log(`Descargando: ${pct.toFixed(1)}%`);
  }
};
xhr.onloadstart = () => console.log('Petición iniciada');
xhr.onloadend   = () => console.log('Petición finalizada (éxito o error)');

// ── Enviar ────────────────────────────────────────────────────────────
xhr.send(); // GET no lleva cuerpo
// xhr.abort(); // cancelar

2.3 XHR — POST y FormData

xhr_post.js
// XHR — Petición POST con JSON, envuelta en Promise
function crearUsuarioXHR(datosUsuario) {
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open('POST', '/api/usuarios');
    xhr.setRequestHeader('Content-Type', 'application/json');
    xhr.responseType = 'json';
    xhr.timeout = 10000;

    xhr.onload = function() {
      if (this.status >= 200 && this.status < 300) {
        resolve(this.response);
      } else {
        reject(new Error(`Error ${this.status}: ${this.statusText}`));
      }
    };
    xhr.onerror  = () => reject(new Error('Error de red'));
    xhr.ontimeout = () => reject(new Error('Timeout'));

    xhr.send(JSON.stringify(datosUsuario)); // body: objeto → string JSON
  });
}

// Uso
crearUsuarioXHR({ nombre: 'Ana', email: 'ana@mail.com' })
  .then(usuario => console.log('Creado:', usuario))
  .catch(error  => console.error('Error:', error.message));

// ── Envío de FormData con XHR ─────────────────────────────────────────
const formulario = document.getElementById('mi-form');
const formData   = new FormData(formulario);
const xhrForm    = new XMLHttpRequest();
xhrForm.open('POST', '/api/datos');
// NO poner Content-Type: multipart/form-data se añade automáticamente
xhrForm.onload = () => console.log(xhrForm.responseText);
xhrForm.send(formData);
