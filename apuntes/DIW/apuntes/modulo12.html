12.1 Axios — librería HTTP popular

Axios es la librería HTTP más popular para JavaScript. Ofrece una sintaxis muy limpia, interceptores, transformación automática de JSON y mejor manejo de errores que Fetch nativo.
axios_completo.js
// Instalar: npm install axios
// CDN: <script src="https://cdn.jsdelivr.net/npm/axios/dist/axios.min.js"></script>

// ── Diferencias clave Axios vs Fetch ──────────────────────────────────
// ✅ Axios rechaza la Promise en errores HTTP (4xx, 5xx) — Fetch NO lo hace
// ✅ Axios parsea JSON automáticamente
// ✅ Axios cancela con AbortController (moderno) o CancelToken (legacy)
// ✅ Axios tiene interceptores para todas las peticiones
// ✅ Axios soporta Node.js de forma nativa

// ── CRUD con Axios ────────────────────────────────────────────────────
const { data: usuarios } = await axios.get('/api/usuarios');
const { data: usuario  } = await axios.get('/api/usuarios/1');

// GET con query params automáticos:
const res = await axios.get('/api/productos', {
  params: { page: 1, per_page: 20, categoria: 'electronica' }
  // → GET /api/productos?page=1&per_page=20&categoria=electronica
});

const { data: nuevo }   = await axios.post('/api/usuarios', { nombre: 'Ana', email: 'ana@mail.com' });
await axios.put('/api/usuarios/1',   { nombre: 'Ana M.' });
await axios.patch('/api/usuarios/1', { email: 'nueva@mail.com' });
await axios.delete('/api/usuarios/1');

// ── Instancia con configuración base ─────────────────────────────────
const api = axios.create({
  baseURL: 'https://mi-api.com/api',
  timeout: 10000,
  headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` },
});

// ── Interceptores ─────────────────────────────────────────────────────
// Añadir token a TODAS las peticiones automáticamente:
api.interceptors.request.use(config => {
  const token = localStorage.getItem('token');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

// Manejar errores 401 globalmente:
api.interceptors.response.use(
  response => response,
  error => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

// ── Manejo de errores con Axios ───────────────────────────────────────
try {
  const res = await api.get('/usuarios/9999');
} catch (error) {
  if (error.response) {
    // Error HTTP (4xx, 5xx) — Axios SÍ rechaza la Promise aquí
    console.error('Estado:', error.response.status);
    console.error('Datos:', error.response.data);
  } else if (error.request) {
    console.error('Sin respuesta del servidor');
  } else {
    console.error('Error:', error.message);
  }
}

12.2 jQuery AJAX ($.ajax, $.get, $.post)

jQuery está en desuso para proyectos nuevos, pero sigue apareciendo en proyectos legacy y en algunos exámenes.
jquery_ajax.js
// jQuery AJAX básico
$.get('/api/usuarios', function(datos) {
  console.log('Usuarios:', datos);
});

$.post('/api/usuarios', { nombre: 'Ana', email: 'ana@mail.com' },
  function(respuesta) { console.log('Creado:', respuesta); }
);

// ── $.ajax — control total ────────────────────────────────────────────
$.ajax({
  url:         '/api/usuarios/1',
  method:      'PUT',
  contentType: 'application/json',
  data:        JSON.stringify({ nombre: 'Ana M.' }),
  dataType:    'json',  // parsear respuesta como JSON
  timeout:     5000,
  beforeSend: function(xhr) {
    xhr.setRequestHeader('Authorization', 'Bearer ' + token);
    $('#btn').prop('disabled', true);
  },
  success: function(datos, status, xhr) {
    console.log('Éxito:', datos);
  },
  error: function(xhr, status, error) {
    console.error('Error:', xhr.status, error);
  },
  complete: function(xhr, status) {
    $('#btn').prop('disabled', false); // siempre se ejecuta
  }
});

// ── jQuery AJAX moderno con Promises ─────────────────────────────────
$.getJSON('/api/usuarios')
  .done(datos => console.log(datos))
  .fail((xhr, status, err) => console.error(err))
  .always(() => console.log('Completado'));
