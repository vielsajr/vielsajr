7.1 Principios REST y diseño de URLs

Principios REST:
• Sin estado (Stateless): cada petición contiene toda la información necesaria. El servidor no guarda estado entre peticiones.
• Recursos: todo se modela como recursos con URLs únicas. /api/usuarios, /api/productos/42
• Verbos HTTP: GET lee, POST crea, PUT/PATCH actualiza, DELETE elimina.
• Representaciones: los recursos se devuelven en JSON (hoy en día casi siempre).
• Interfaz uniforme: URLs consistentes, cabeceras estándar, códigos HTTP correctos.
• Capas: el cliente no sabe si habla con el servidor real o un intermediario (proxy, CDN...).

7.2 Clase ApiClient reutilizable

api_client.js
// Clase ApiClient — abstracción reutilizable para consumir APIs REST
class ApiClient {
  #baseUrl;
  #token;

  constructor(baseUrl, token = null) {
    this.#baseUrl = baseUrl.replace(/\/$/, '');
    this.#token   = token;
  }

  setToken(token) { this.#token = token; }

  #getHeaders(extra = {}) {
    const headers = {
      'Content-Type': 'application/json',
      'Accept':       'application/json',
      ...extra,
    };
    if (this.#token) headers['Authorization'] = `Bearer ${this.#token}`;
    return headers;
  }

  async #request(method, endpoint, data = null, signal = null) {
    const url    = `${this.#baseUrl}${endpoint}`;
    const config = { method, headers: this.#getHeaders() };

    if (data)   config.body   = JSON.stringify(data);
    if (signal) config.signal = signal;

    const response = await fetch(url, config);
    if (!response.ok) {
      let errorMsg = `Error ${response.status}`;
      try {
        const err = await response.json();
        errorMsg = err.mensaje || err.error || errorMsg;
      } catch (_) {}
      throw new Error(errorMsg);
    }
    if (response.status === 204) return null;
    return response.json();
  }

  get(endpoint, signal)         { return this.#request('GET',    endpoint, null, signal); }
  post(endpoint, data, signal)  { return this.#request('POST',   endpoint, data, signal); }
  put(endpoint, data, signal)   { return this.#request('PUT',    endpoint, data, signal); }
  patch(endpoint, data, signal) { return this.#request('PATCH',  endpoint, data, signal); }
  delete(endpoint, signal)      { return this.#request('DELETE', endpoint, null, signal); }
}

// ── Uso del ApiClient ─────────────────────────────────────────────────
const api = new ApiClient('https://mi-api.com/api', localStorage.getItem('token'));

const productos = await api.get('/productos?page=1&per_page=20');
const producto  = await api.get('/productos/42');
const nuevo     = await api.post('/productos', { nombre: 'Teclado', precio: 89.99 });
const actualiz  = await api.patch('/productos/42', { precio: 79.99 });
await api.delete('/productos/42');
