3.1 Fetch vs XHR y anatomía completa

Fetch vs XHR — diferencias clave:
• Fetch devuelve Promises → compatible con .then()/.catch() y async/await
• Fetch NO rechaza la Promise en errores HTTP (404, 500…) — solo rechaza si hay error de red. Hay que comprobar response.ok
• Fetch tiene mejor integración con APIs modernas (Streams, Cache API, Service Workers)
• XHR tiene mejor soporte para progreso de subida (upload progress)
• XHR se cancela con xhr.abort(). Fetch necesita AbortController
• XHR soporta peticiones síncronas (bloqueantes). Fetch es siempre asíncrono
fetch_anatomia.js
// fetch(url, opciones?) devuelve Promise<Response>

fetch('https://jsonplaceholder.typicode.com/users/1', {
  method: 'GET',        // 'POST' | 'PUT' | 'PATCH' | 'DELETE'
  headers: {
    'Content-Type': 'application/json',
    'Accept':       'application/json',
    'Authorization':'Bearer mi-token',
  },
  // body: JSON.stringify({ nombre: 'Ana' }), // solo POST, PUT, PATCH
  cache:          'no-cache',  // 'default'|'no-store'|'reload'|'no-cache'|'force-cache'
  credentials:    'same-origin', // 'omit'|'same-origin'|'include'
  redirect:       'follow',    // 'follow'|'error'|'manual'
  mode:           'cors',      // 'cors'|'no-cors'|'same-origin'
  // signal: miAbortController.signal,
})
.then(response => {
  // ── El objeto Response ─────────────────────────────────────────────
  console.log(response.ok);         // true si status 200-299
  console.log(response.status);     // 200, 404, 500...
  console.log(response.statusText); // 'OK', 'Not Found'...
  console.log(response.headers.get('Content-Type'));
  console.log(response.url);
  console.log(response.redirected);

  // ⚠️ El body solo se puede leer UNA vez — elige uno:
  return response.json();       // parsea como JSON → objeto JS
  // return response.text();    // como string
  // return response.blob();    // datos binarios (imagen, PDF...)
  // return response.arrayBuffer();
})
.then(datos  => console.log('Datos:', datos))
.catch(error => console.error('Error de red:', error.message));

3.2 response.ok — el error más común

fetch_response_ok.js
// ── MALO: no comprueba errores HTTP ──────────────────────────────────
fetch('/api/usuarios/9999') // devuelve 404
  .then(res  => res.json())       // ← ejecuta aunque sea 404
  .then(datos => console.log(datos)) // ← puede dar error raro
  .catch(err  => console.error(err)); // solo captura errores de red

// ── BIEN: comprobación correcta con response.ok ───────────────────────
fetch('/api/usuarios/1')
  .then(response => {
    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }
    return response.json();
  })
  .then(usuario => console.log('Usuario:', usuario.name))
  .catch(error  => console.error('Error:', error.message));

// ── MEJOR: función reutilizable ───────────────────────────────────────
async function fetchJSON(url, opciones = {}) {
  const response = await fetch(url, opciones);
  if (!response.ok) {
    let mensajeError = `Error ${response.status}`;
    try {
      const errorData = await response.json();
      mensajeError = errorData.mensaje || errorData.error || mensajeError;
    } catch (e) {}
    throw new Error(mensajeError);
  }
  if (response.status === 204) return null; // No Content
  return response.json();
}

3.3 Todos los métodos HTTP con Fetch

fetch_metodos.js
const API = 'https://jsonplaceholder.typicode.com';

// ── GET ───────────────────────────────────────────────────────────────
const getUsuarios = () =>
  fetch(`${API}/users`).then(r => r.ok ? r.json() : Promise.reject(r.status));

// GET con query params
const buscar = (termino) => {
  const params = new URLSearchParams({ q: termino, _limit: 10 });
  return fetch(`${API}/users?${params}`).then(r => r.json());
};

// ── POST ──────────────────────────────────────────────────────────────
const crearPost = (datos) =>
  fetch(`${API}/posts`, {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify(datos),
  }).then(r => r.json());

// ── PUT (reemplaza todo el recurso) ───────────────────────────────────
const actualizarPost = (id, datos) =>
  fetch(`${API}/posts/${id}`, {
    method: 'PUT', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(datos),
  }).then(r => r.json());

// ── PATCH (actualiza parcialmente) ────────────────────────────────────
const patchPost = (id, cambios) =>
  fetch(`${API}/posts/${id}`, {
    method: 'PATCH', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(cambios),
  }).then(r => r.json());

// ── DELETE ────────────────────────────────────────────────────────────
const eliminarPost = (id) =>
  fetch(`${API}/posts/${id}`, { method: 'DELETE' })
    .then(r => { if (!r.ok) throw new Error(`Error ${r.status}`); return true; });

// ── Subir archivo (FormData) ──────────────────────────────────────────
const subirImagen = (archivo) => {
  const fd = new FormData();
  fd.append('imagen', archivo);
  fd.append('descripcion', 'Mi foto');
  // NO poner Content-Type: el navegador lo gestiona (con boundary)
  return fetch('/api/imagenes', { method: 'POST', body: fd }).then(r => r.json());
};

3.4 Cancelar peticiones con AbortController

abort_controller.js
// AbortController permite cancelar peticiones Fetch en curso
const controller = new AbortController();
const signal = controller.signal;

fetch('/api/datos-lentos', { signal })
  .then(r => r.json())
  .then(datos => console.log(datos))
  .catch(err => {
    if (err.name === 'AbortError') {
      console.log('Petición cancelada');
    } else {
      console.error('Error:', err.message);
    }
  });

// Cancelar después de 3 segundos (timeout manual)
setTimeout(() => controller.abort(), 3000);

// ── Patrón: cancelar búsqueda anterior al escribir ────────────────────
let abortController = null;

document.getElementById('busqueda').addEventListener('input', async (e) => {
  if (abortController) abortController.abort(); // cancelar anterior
  abortController = new AbortController();

  const termino = e.target.value.trim();
  if (termino.length < 2) return;

  try {
    const res = await fetch(`/api/buscar?q=${termino}`, {
      signal: abortController.signal
    });
    const resultados = await res.json();
    mostrarResultados(resultados);
  } catch (err) {
    if (err.name !== 'AbortError') console.error('Error:', err.message);
  }
});
