9.1 Tipos de errores en AJAX

Tipo de errorDescripción y cómo detectarlo
Error de redSin conexión, servidor caído, timeout. TypeError: Failed to fetch. La Promise se rechaza.
Error CORSEl servidor no envía las cabeceras CORS correctas. TypeError en la consola.
Error HTTP 4xxError del cliente. 400, 401, 403, 404, 422... La Promise NO se rechaza en Fetch.
Error HTTP 5xxError del servidor. La Promise NO se rechaza — hay que comprobar response.ok.
Error de parseo JSONEl servidor devuelve HTML o texto en vez de JSON. JSON.parse() lanza SyntaxError.
AbortErrorLa petición fue cancelada con AbortController. error.name === "AbortError".
Timeout propioImplementado con AbortController + setTimeout.

9.2 Fetch robusto con timeout y notificaciones

fetch_robusto.js
// Función de fetch robusta con timeout, manejo completo de errores
async function fetchRobusto(url, opciones = {}, timeoutMs = 10000) {
  const controller = new AbortController();
  const timerId = setTimeout(() => controller.abort(), timeoutMs);

  try {
    const response = await fetch(url, {
      ...opciones,
      signal: controller.signal,
    });
    clearTimeout(timerId);

    // Intentar parsear el body (éxito o error)
    const contentType = response.headers.get('Content-Type') || '';
    let body = null;
    if (contentType.includes('application/json')) {
      body = await response.json().catch(() => null);
    } else {
      body = await response.text().catch(() => '');
    }

    if (!response.ok) {
      const mensaje = (body && body.mensaje) || (body && body.error)
        || `Error ${response.status}: ${response.statusText}`;
      const error = new Error(mensaje);
      error.status = response.status;
      error.body   = body;
      throw error;
    }
    return body;

  } catch (error) {
    clearTimeout(timerId);
    if (error.name === 'AbortError') {
      throw new Error(`Timeout: la petición tardó más de ${timeoutMs}ms`);
    }
    if (error.name === 'TypeError') {
      throw new Error('Error de red: comprueba tu conexión a internet');
    }
    throw error;
  }
}

// ── Sistema de notificaciones de error ────────────────────────────────
function mostrarError(error) {
  const div = document.getElementById('mensajes');
  let mensaje = error.message;
  let tipo    = 'error';

  if (error.status === 401) {
    mensaje = 'Sesión caducada. Por favor, vuelve a iniciar sesión.';
    redirigirLogin();
  } else if (error.status === 403) {
    mensaje = 'No tienes permisos para realizar esta acción.';
  } else if (error.status === 404) {
    mensaje = 'El recurso solicitado no existe.';
    tipo = 'warning';
  } else if (error.status === 422) {
    const erroresValidacion = error.body?.errores || {};
    mensaje = Object.values(erroresValidacion).flat().join(', ');
  } else if (error.status >= 500) {
    mensaje = 'Error del servidor. Por favor, inténtalo más tarde.';
  }

  div.innerHTML = `<div class="alert alert-${tipo}">${mensaje}</div>`;
  setTimeout(() => div.innerHTML = '', 5000);
}
