13.1 App de Notas: backend PHP

api/notas.php
<?php
// api/notas.php — backend PHP completo para la app de notas
header('Content-Type: application/json; charset=UTF-8');
header('Access-Control-Allow-Origin: *');
header('Access-Control-Allow-Methods: GET, POST, PATCH, DELETE, OPTIONS');
header('Access-Control-Allow-Headers: Content-Type');

if ($_SERVER['REQUEST_METHOD'] === 'OPTIONS') { http_response_code(204); exit; }

require '../includes/db.php';
$pdo    = getDB();
$method = $_SERVER['REQUEST_METHOD'];
$id     = isset($_GET['id']) ? (int)$_GET['id'] : null;

function resp($data, $code = 200) {
  http_response_code($code);
  echo json_encode($data, JSON_UNESCAPED_UNICODE);
  exit;
}
function body() {
  return json_decode(file_get_contents('php://input'), true) ?? [];
}

switch ($method) {
  case 'GET':
    if ($id) {
      $stmt = $pdo->prepare('SELECT * FROM notas WHERE id = ?');
      $stmt->execute([$id]);
      $nota = $stmt->fetch();
      $nota ? resp($nota) : resp(['error' => 'No encontrada'], 404);
    } else {
      resp($pdo->query('SELECT * FROM notas ORDER BY creada_en DESC')->fetchAll());
    }
    break;

  case 'POST':
    $b = body();
    if (empty($b['titulo'])) resp(['error' => 'Título obligatorio'], 422);
    $st = $pdo->prepare('INSERT INTO notas (titulo, contenido) VALUES (?,?)');
    $st->execute([htmlspecialchars($b['titulo']), htmlspecialchars($b['contenido'] ?? '')]);
    resp($pdo->query('SELECT * FROM notas WHERE id = LAST_INSERT_ID()')->fetch(), 201);
    break;

  case 'PATCH':
    if (!$id) resp(['error' => 'ID requerido'], 400);
    $b  = body();
    $st = $pdo->prepare('UPDATE notas SET titulo=COALESCE(?,titulo), contenido=COALESCE(?,contenido) WHERE id=?');
    $st->execute([$b['titulo'] ?? null, $b['contenido'] ?? null, $id]);
    resp(['ok' => true]);
    break;

  case 'DELETE':
    if (!$id) resp(['error' => 'ID requerido'], 400);
    $pdo->prepare('DELETE FROM notas WHERE id=?')->execute([$id]);
    resp(null, 204);
    break;

  default: resp(['error' => 'Método no permitido'], 405);
}
?>

13.2 App de Notas: frontend JavaScript

notas_app.js
// notas_app.js — frontend completo de la app de notas
const API       = '/api/notas.php';
const listaEl   = document.getElementById('lista-notas');
const formEl    = document.getElementById('form-nota');
const tituloEl  = document.getElementById('titulo');
const contenido = document.getElementById('contenido');
const editIdEl  = document.getElementById('edit-id');
const btnCancel = document.getElementById('btn-cancelar');

// ── Cargar todas las notas ─────────────────────────────────────────────
async function cargar() {
  listaEl.innerHTML = '<p>Cargando...</p>';
  const notas = await fetch(API).then(r => r.json());
  listaEl.innerHTML = '';
  if (notas.length === 0) {
    listaEl.innerHTML = '<p class="vacio">No tienes notas aún.</p>';
    return;
  }
  notas.forEach(nota => listaEl.appendChild(renderNota(nota)));
}

// ── Renderizar una nota ───────────────────────────────────────────────
function renderNota(nota) {
  const div = document.createElement('div');
  div.className = 'nota-card';
  div.dataset.id = nota.id;
  div.innerHTML = `
    <h3>${nota.titulo}</h3>
    <p>${nota.contenido || '<em>Sin contenido</em>'}</p>
    <small>${new Date(nota.creada_en).toLocaleString('es-ES')}</small>
    <div class="acciones">
      <button class="btn-editar"  data-id="${nota.id}">✏️ Editar</button>
      <button class="btn-borrar"  data-id="${nota.id}">🗑️ Borrar</button>
    </div>
  `;
  div.querySelector('.btn-editar').addEventListener('click', () => editarNota(nota));
  div.querySelector('.btn-borrar').addEventListener('click', () => borrarNota(nota.id, div));
  return div;
}

// ── Crear o actualizar ────────────────────────────────────────────────
formEl.addEventListener('submit', async (e) => {
  e.preventDefault();
  const id     = editIdEl.value;
  const body   = { titulo: tituloEl.value, contenido: contenido.value };
  const url    = id ? `${API}?id=${id}` : API;
  const method = id ? 'PATCH' : 'POST';
  await fetch(url, {
    method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body)
  });
  formEl.reset(); editIdEl.value = ''; btnCancel.style.display = 'none';
  await cargar();
});

// ── Editar (rellenar formulario) ──────────────────────────────────────
function editarNota(nota) {
  tituloEl.value  = nota.titulo;
  contenido.value = nota.contenido || '';
  editIdEl.value  = nota.id;
  btnCancel.style.display = 'inline';
  tituloEl.focus();
}

// ── Borrar ────────────────────────────────────────────────────────────
async function borrarNota(id, elemento) {
  if (!confirm('¿Eliminar esta nota?')) return;
  await fetch(`${API}?id=${id}`, { method: 'DELETE' });
  elemento.remove();
}

btnCancel.addEventListener('click', () => {
  formEl.reset(); editIdEl.value = ''; btnCancel.style.display = 'none';
});

cargar();
