4.1 Crear y consumir Promises

Una Promise puede estar en tres estados: pending (pendiente), fulfilled (resuelta con éxito) o rejected (rechazada con error). Una vez resuelta o rechazada, no puede cambiar de estado.
promises_basico.js
// ════════════════════════════════════════════════════════════
// CREAR una Promise
// ════════════════════════════════════════════════════════════
const miPromesa = new Promise((resolve, reject) => {
  const exito = Math.random() > 0.5;
  setTimeout(() => {
    if (exito) resolve({ datos: 'Todo bien' }); // → fulfilled
    else       reject(new Error('Algo falló'));  // → rejected
  }, 1000);
});

// ════════════════════════════════════════════════════════════
// CONSUMIR una Promise
// ════════════════════════════════════════════════════════════
miPromesa
  .then(resultado => {
    console.log('Éxito:', resultado.datos);
    return resultado.datos.toUpperCase(); // ← encadenar
  })
  .then(textoMayus => {
    console.log('Transformado:', textoMayus); // recibe el return anterior
  })
  .catch(error => {
    console.error('Error capturado:', error.message);
  })
  .finally(() => {
    // Se ejecuta SIEMPRE (éxito o error) — ideal para ocultar spinners
    console.log('Operación finalizada');
    ocultarSpinner();
  });

// ── Promesas ya resueltas/rechazadas ──────────────────────────────────
Promise.resolve(42);              // Promise ya resuelta con 42
Promise.reject(new Error('Fallo')); // Promise ya rechazada

4.2 Métodos estáticos de Promise

promise_estaticos.js
// ════════════════════════════════════════════════════════════
// Promise.all — esperar TODAS (falla si falla UNA)
// ════════════════════════════════════════════════════════════
const p1 = fetch('/api/usuarios').then(r => r.json());
const p2 = fetch('/api/productos').then(r => r.json());
const p3 = fetch('/api/pedidos').then(r => r.json());

Promise.all([p1, p2, p3])
  .then(([usuarios, productos, pedidos]) => {
    // Recibe array con los resultados EN EL MISMO ORDEN
    console.log(usuarios.length, productos.length, pedidos.length);
  })
  .catch(error => {
    // Si CUALQUIERA falla, se rechaza con ese error
    console.error('Una de las peticiones falló:', error.message);
  });

// ════════════════════════════════════════════════════════════
// Promise.allSettled — esperar TODAS aunque fallen (nunca rechaza)
// ════════════════════════════════════════════════════════════
Promise.allSettled([p1, p2, p3])
  .then(resultados => {
    resultados.forEach((resultado, i) => {
      if (resultado.status === 'fulfilled') {
        console.log(`Petición ${i} OK:`, resultado.value);
      } else {
        console.error(`Petición ${i} falló:`, resultado.reason.message);
      }
    });
  });

// ════════════════════════════════════════════════════════════
// Promise.race — la PRIMERA que termine (éxito O error)
// ════════════════════════════════════════════════════════════
const timeout = new Promise((_, reject) =>
  setTimeout(() => reject(new Error('Timeout 5s')), 5000)
);
Promise.race([fetch('/api/lento'), timeout])
  .then(r => r.json())
  .catch(err => console.error(err.message)); // 'Timeout 5s' si tarda más

// ════════════════════════════════════════════════════════════
// Promise.any — la PRIMERA con ÉXITO (ignora rechazos)
// Falla solo si TODAS fallan → AggregateError
// ════════════════════════════════════════════════════════════
Promise.any([
  fetch('https://servidor1.com/api'),
  fetch('https://servidor2.com/api'),
  fetch('https://servidor3.com/api'),
])
  .then(res => res.json())
  .then(datos => console.log('Respuesta del más rápido:', datos))
  .catch(err  => console.error('Todos fallaron:', err));

4.3 Encadenamiento y transformación

encadenamiento.js
// Cada .then() devuelve una NUEVA Promise — permite transformar y encadenar
fetch('/api/usuarios/1')
  .then(response => {
    if (!response.ok) throw new Error(`Error ${response.status}`);
    return response.json();
  })
  .then(usuario => {
    console.log('Usuario obtenido:', usuario.name);
    // Podemos retornar OTRA Promise para continuar la cadena
    return fetch(`/api/pedidos?userId=${usuario.id}`);
  })
  .then(response => response.json())
  .then(pedidos => {
    console.log('Pedidos del usuario:', pedidos.length);
    return pedidos.filter(p => p.estado === 'pendiente');
  })
  .then(pendientes => console.log('Pendientes:', pendientes.length))
  .catch(error => {
    // UN SOLO .catch() captura errores de CUALQUIER paso anterior
    console.error('Error en algún paso:', error.message);
  })
  .finally(() => {
    document.getElementById('spinner').style.display = 'none';
  });
