3.1 Ciclo de vida

Estados de un contenedor:
created → Creado pero no iniciado  |  running → En ejecución  |  paused → Procesos suspendidos  |  stopped → Parado (existe)  |  removed → Eliminado definitivamente

3.2 Comandos esenciales de contenedores

Comandos de contenedores
# ── Crear y ejecutar ────────────────────────────────────────────
docker run [opciones] IMAGEN [COMANDO]

docker run -d nginx                        # -d detached (background)
docker run -it ubuntu bash                 # -i interactivo, -t TTY
docker run --name mi-nginx nginx           # nombre personalizado
docker run -p 8080:80 nginx                # -p host:contenedor
docker run -p 127.0.0.1:8080:80 nginx      # solo accesible desde localhost
docker run -P nginx                        # mapea todos los EXPOSE a puertos aleatorios
docker run -v /host/path:/container/path nginx  # bind mount
docker run -v mi-volumen:/data nginx       # volumen nombrado
docker run -e MYSQL_ROOT_PASSWORD=pass mysql    # variable de entorno
docker run --env-file .env nginx           # variables desde archivo
docker run --rm nginx                      # eliminar al parar
docker run --network mi-red nginx          # red específica
docker run --memory='512m' --cpus='0.5' nginx   # límites de recursos
docker run --restart=always nginx          # política de reinicio
docker run --hostname mi-host nginx        # hostname del contenedor
docker run -w /app nginx                   # working directory
docker run -u 1000:1000 nginx              # usuario:grupo

# ── Gestión básica ────────────────────────────────────────────
docker ps                                  # contenedores en ejecución
docker ps -a                               # todos (incluidos parados)
docker ps --format 'table {{.Names}}	{{.Status}}	{{.Ports}}'
docker start mi-nginx
docker stop mi-nginx                       # SIGTERM → espera → SIGKILL
docker stop -t 5 mi-nginx                  # timeout de 5 segundos
docker kill mi-nginx                       # forzar parada (SIGKILL)
docker restart mi-nginx
docker pause mi-nginx
docker unpause mi-nginx
docker rm mi-nginx                         # eliminar contenedor parado
docker rm -f mi-nginx                      # forzar eliminación
docker container prune                     # eliminar todos los parados

# ── Inspección y logs ─────────────────────────────────────────
docker logs mi-nginx                       # ver logs
docker logs -f mi-nginx                    # seguir en tiempo real
docker logs --tail 100 mi-nginx            # últimas 100 líneas
docker logs --since 1h mi-nginx            # desde hace 1 hora
docker inspect mi-nginx                    # info detallada (JSON)
docker stats                               # uso de recursos en tiempo real
docker stats mi-nginx --no-stream          # una sola lectura
docker top mi-nginx                        # procesos del contenedor
docker port mi-nginx                       # mapeos de puertos

# ── Ejecutar comandos en contenedor en marcha ─────────────────
docker exec mi-nginx ls /var/www/html
docker exec -it mi-nginx bash              # abrir shell interactiva
docker exec -it mi-nginx sh                # con sh (si bash no está)
docker exec -u root mi-nginx whoami        # como root

# ── Copiar archivos ───────────────────────────────────────────
docker cp index.html mi-nginx:/var/www/html/
docker cp mi-nginx:/var/log/nginx/error.log ./logs/

# ── Limpieza general ──────────────────────────────────────────
docker system prune                        # contenedores, redes, imágenes sin usar
docker system prune -a --volumes           # TODO incluyendo volúmenes
docker system df                           # uso de espacio en disco
