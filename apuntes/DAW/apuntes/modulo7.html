7.1 Stack LAMP

docker-compose.yml — LAMP
services:

  # ── Servidor Web + PHP ────────────────────────────────────────
  web:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: lamp-web
    ports:
      - '8080:80'
    volumes:
      - ./src:/var/www/html
      - ./docker/php/php.ini:/usr/local/etc/php/conf.d/custom.ini
    environment:
      DB_HOST: db
      DB_PORT: 3306
      DB_NAME: mi_base_datos
      DB_USER: usuario
      DB_PASS: mi_password
    depends_on:
      db:
        condition: service_healthy
    networks:
      - lamp-network
    restart: unless-stopped

  # ── Base de datos MySQL ────────────────────────────────────────
  db:
    image: mysql:8.0
    container_name: lamp-db
    environment:
      MYSQL_ROOT_PASSWORD: root_password
      MYSQL_DATABASE: mi_base_datos
      MYSQL_USER: usuario
      MYSQL_PASSWORD: mi_password
    volumes:
      - mysql-data:/var/lib/mysql
      - ./docker/mysql/init.sql:/docker-entrypoint-initdb.d/init.sql
    ports:
      - '3306:3306'                        # quitar en producción
    healthcheck:
      test: ['CMD', 'mysqladmin', 'ping', '-h', 'localhost', '-uroot', '-proot_password']
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    networks:
      - lamp-network
    restart: unless-stopped

  # ── phpMyAdmin ─────────────────────────────────────────────────
  phpmyadmin:
    image: phpmyadmin:5.2
    container_name: lamp-phpmyadmin
    environment:
      PMA_HOST: db
      PMA_PORT: 3306
      PMA_USER: root
      PMA_PASSWORD: root_password
    ports:
      - '8081:80'
    depends_on:
      - db
    networks:
      - lamp-network
    profiles:
      - tools                              # solo con: docker compose --profile tools up

volumes:
  mysql-data:

networks:
  lamp-network:
    driver: bridge

7.2 Stack LEMP (Nginx + PHP-FPM + MySQL + Redis)

LEMP usa Nginx como servidor web (en lugar de Apache) con PHP-FPM como procesador PHP. Es el stack más usado en producción por su rendimiento.
docker-compose.yml — LEMP
services:
  nginx:
    image: nginx:1.25-alpine
    container_name: lemp-nginx
    ports:
      - '80:80'
      - '443:443'
    volumes:
      - ./src:/var/www/html
      - ./docker/nginx/conf.d:/etc/nginx/conf.d
      - ./docker/nginx/ssl:/etc/nginx/ssl
    depends_on:
      - php
    networks:
      - lemp-network
    restart: unless-stopped

  php:
    build:
      context: ./docker/php
    container_name: lemp-php
    volumes:
      - ./src:/var/www/html
    environment:
      DB_HOST: mysql
      REDIS_HOST: redis
    depends_on:
      mysql:
        condition: service_healthy
    networks:
      - lemp-network
    restart: unless-stopped

  mysql:
    image: mysql:8.0
    container_name: lemp-mysql
    env_file: .env
    volumes:
      - mysql-data:/var/lib/mysql
    healthcheck:
      test: ['CMD', 'mysqladmin', 'ping', '-h', 'localhost']
      interval: 10s
      retries: 5
    networks:
      - lemp-network
    restart: unless-stopped

  redis:
    image: redis:7-alpine
    container_name: lemp-redis
    volumes:
      - redis-data:/data
    networks:
      - lemp-network
    restart: unless-stopped

volumes:
  mysql-data:
  redis-data:

networks:
  lemp-network:
docker/nginx/conf.d/default.conf
server {
    listen 80;
    server_name localhost;
    root /var/www/html/public;
    index index.php index.html;

    access_log /var/log/nginx/access.log;
    error_log  /var/log/nginx/error.log;

    location / {
        try_files $uri $uri/ /index.php?$query_string;
    }

    location ~ \.php$ {
        fastcgi_pass php:9000;            # nombre del servicio PHP-FPM
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;
        fastcgi_read_timeout 300;
    }

    location ~ /\.ht {
        deny all;
    }
}

7.3 Stack Laravel completo

docker-compose.yml — Laravel
services:
  app:
    build:
      context: .
      target: development
    container_name: laravel-app
    volumes:
      - .:/var/www/html
      - /var/www/html/vendor           # volumen anónimo para vendor/
    environment:
      APP_ENV: local
      APP_DEBUG: 'true'
      DB_CONNECTION: mysql
      DB_HOST: mysql
      DB_PORT: 3306
      DB_DATABASE: laravel_db
      DB_USERNAME: laravel
      DB_PASSWORD: secret
      REDIS_HOST: redis
      CACHE_DRIVER: redis
      SESSION_DRIVER: redis
      QUEUE_CONNECTION: redis
    depends_on:
      mysql:
        condition: service_healthy
    networks:
      - laravel-network

  nginx:
    image: nginx:1.25-alpine
    container_name: laravel-nginx
    ports:
      - '8000:80'
    volumes:
      - .:/var/www/html
      - ./docker/nginx/default.conf:/etc/nginx/conf.d/default.conf
    depends_on:
      - app
    networks:
      - laravel-network

  mysql:
    image: mysql:8.0
    container_name: laravel-mysql
    environment:
      MYSQL_ROOT_PASSWORD: root
      MYSQL_DATABASE: laravel_db
      MYSQL_USER: laravel
      MYSQL_PASSWORD: secret
    volumes:
      - mysql-data:/var/lib/mysql
    healthcheck:
      test: ['CMD', 'mysqladmin', 'ping', '-h', 'localhost', '-uroot', '-proot']
      interval: 10s
      retries: 5
    networks:
      - laravel-network

  redis:
    image: redis:7-alpine
    networks:
      - laravel-network

  queue:
    build: .
    container_name: laravel-queue
    command: php artisan queue:work --tries=3
    volumes:
      - .:/var/www/html
    depends_on:
      - mysql
      - redis
    networks:
      - laravel-network
    restart: unless-stopped

  scheduler:
    build: .
    container_name: laravel-scheduler
    command: sh -c 'while true; do php artisan schedule:run --verbose --no-interaction; sleep 60; done'
    volumes:
      - .:/var/www/html
    depends_on:
      - mysql
    networks:
      - laravel-network
    restart: unless-stopped

volumes:
  mysql-data:

networks:
  laravel-network:
