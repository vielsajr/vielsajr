6.1 Comandos Docker Compose

Docker Compose v2 viene integrado en Docker Desktop y como plugin en Linux (docker compose, sin guion). La versión v1 (docker-compose, con guion) está deprecada.
Comandos Docker Compose
# Levantar todos los servicios (en background)
docker compose up -d

# Levantar y reconstruir si hay cambios en Dockerfile
docker compose up -d --build

# Escalar servicio (múltiples instancias)
docker compose up -d --scale web=3

# Parar y eliminar contenedores, redes
docker compose down

# Parar y eliminar también volúmenes
docker compose down -v

# Solo parar (sin eliminar)
docker compose stop
docker compose start

# Ver logs de todos los servicios
docker compose logs -f
docker compose logs -f web             # solo servicio 'web'
docker compose logs --tail 50

# Estado de los servicios
docker compose ps
docker compose top

# Ejecutar comando en servicio
docker compose exec web bash
docker compose exec db mysql -u root -p

# Reconstruir imagen
docker compose build
docker compose build --no-cache web

# Usar archivo compose alternativo
docker compose -f docker-compose.prod.yml up -d

# Combinar archivos (base + override)
docker compose -f docker-compose.yml -f docker-compose.override.yml up -d

# Ver la configuración resultante (merge)
docker compose config

# Reiniciar un servicio
docker compose restart web

# Eliminar imágenes construidas
docker compose down --rmi all

6.2 Referencia completa docker-compose.yml

docker-compose.yml
# Formato actual no requiere 'version' (Compose Spec)

services:
  nombre-servicio:

    # ── Imagen ───────────────────────────────────────────────
    image: nginx:1.25                      # imagen del registro
    # O construir desde Dockerfile:
    build:
      context: ./app
      dockerfile: Dockerfile.prod
      args:
        NODE_ENV: production
      target: production                   # stage en multi-stage build

    # ── Nombre y hostname ────────────────────────────────────
    container_name: mi-nginx
    hostname: web-server

    # ── Puertos ──────────────────────────────────────────────
    ports:
      - '8080:80'
      - '443:443'
      - '127.0.0.1:8080:80'               # solo localhost

    # ── Variables de entorno ─────────────────────────────────
    environment:
      MYSQL_ROOT_PASSWORD: secret
      APP_ENV: production
    env_file:
      - .env
      - .env.local

    # ── Volúmenes ────────────────────────────────────────────
    volumes:
      - datos-mysql:/var/lib/mysql         # volumen nombrado
      - ./src:/var/www/html                # bind mount
      - ./nginx.conf:/etc/nginx/nginx.conf:ro

    # ── Redes ────────────────────────────────────────────────
    networks:
      - frontend
      - backend

    # ── Dependencias ─────────────────────────────────────────
    depends_on:
      db:
        condition: service_healthy         # esperar a que esté sano
      redis:
        condition: service_started

    # ── Reinicio ─────────────────────────────────────────────
    restart: unless-stopped
    # opciones: no | always | on-failure | unless-stopped

    # ── Healthcheck ──────────────────────────────────────────
    healthcheck:
      test: ['CMD', 'curl', '-f', 'http://localhost/health']
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s

    # ── Recursos ─────────────────────────────────────────────
    deploy:
      resources:
        limits:
          cpus: '0.5'
          memory: 512M
        reservations:
          memory: 128M

    # ── Perfil (no iniciar por defecto) ──────────────────────
    profiles:
      - debug

    # ── Logging ──────────────────────────────────────────────
    logging:
      driver: json-file
      options:
        max-size: '10m'
        max-file: '3'

# ── Definición de volúmenes ──────────────────────────────────
volumes:
  datos-mysql:
    driver: local
  datos-redis:
    external: true                         # volumen ya existente

# ── Definición de redes ───────────────────────────────────────
networks:
  frontend:
    driver: bridge
  backend:
    driver: bridge
    internal: true                         # sin acceso a internet
