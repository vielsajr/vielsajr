9.1 Tests unitarios de servicio con Mockito

UsuarioServiceTest.java 
// UsuarioServiceTest.java
@ExtendWith(MockitoExtension.class)  // JUnit 5 + Mockito
class UsuarioServiceTest {

    @Mock
    private UsuarioRepository repo;

    @Mock
    private PasswordEncoder passwordEncoder;

    @InjectMocks
    private UsuarioServiceImpl service;  // inyecta los mocks automáticamente

    @Test
    @DisplayName("Debe retornar el usuario cuando existe")
    void obtenerPorId_existente() {
        // GIVEN
        Long id = 1L;
        Usuario user = Usuario.builder().id(id).nombre("Ana").email("ana@mail.com").build();
        when(repo.findById(id)).thenReturn(Optional.of(user));

        // WHEN
        UsuarioDTO resultado = service.obtenerPorId(id);

        // THEN
        assertNotNull(resultado);
        assertEquals("Ana", resultado.nombre());
        assertEquals("ana@mail.com", resultado.email());
        verify(repo, times(1)).findById(id);
    }

    @Test
    @DisplayName("Debe lanzar excepción cuando el usuario no existe")
    void obtenerPorId_noExistente() {
        // GIVEN
        when(repo.findById(99L)).thenReturn(Optional.empty());

        // WHEN + THEN
        assertThrows(RecursoNoEncontradoException.class,
            () -> service.obtenerPorId(99L));
    }

    @Test
    void crear_emailDuplicado_lanzaConflicto() {
        UsuarioRequest req = new UsuarioRequest("Ana","ana@mail.com","Pass1234",25,null,null,null);
        when(repo.existsByEmail("ana@mail.com")).thenReturn(true);

        assertThrows(ConflictoException.class, () -> service.crear(req));
        verify(repo, never()).save(any());  // nunca debe llamar a save()
    }
}
Patrón Given/When/Then: separa la preparación, la ejecución y la verificación del test. verify() comprueba que los mocks fueron llamados correctamente.

9.2 Tests de integración del controlador con MockMvc

UsuarioControllerTest.java 
// UsuarioControllerTest.java
@WebMvcTest(UsuarioController.class)  // carga solo el contexto web
class UsuarioControllerTest {

    @Autowired
    private MockMvc mockMvc;

    @MockBean  // mock en el contexto de Spring
    private UsuarioService usuarioService;

    @Autowired
    private ObjectMapper objectMapper;

    @Test
    void getUsuarios_debeRetornar200() throws Exception {
        // GIVEN
        List<UsuarioDTO> lista = List.of(
            new UsuarioDTO(1L, "Ana", "ana@mail.com"),
            new UsuarioDTO(2L, "Luis", "luis@mail.com")
        );
        when(usuarioService.listarTodos(any())).thenReturn(new PageImpl<>(lista));

        // WHEN + THEN
        mockMvc.perform(get("/api/usuarios")
            .contentType(MediaType.APPLICATION_JSON))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.content").isArray())
            .andExpect(jsonPath("$.content[0].nombre").value("Ana"))
            .andExpect(jsonPath("$.totalElements").value(2))
            .andDo(print());  // imprime la petición/respuesta en consola
    }

    @Test
    void crearUsuario_valido_debeRetornar201() throws Exception {
        UsuarioRequest req = new UsuarioRequest("Ana","ana@mail.com","Pass1234",25,null,null,null);
        UsuarioDTO dto = new UsuarioDTO(1L, "Ana", "ana@mail.com");
        when(usuarioService.crear(any())).thenReturn(dto);

        mockMvc.perform(post("/api/usuarios")
            .contentType(MediaType.APPLICATION_JSON)
            .content(objectMapper.writeValueAsString(req)))
            .andExpect(status().isCreated())
            .andExpect(jsonPath("$.id").value(1))
            .andExpect(jsonPath("$.nombre").value("Ana"));
    }

    @Test
    void crearUsuario_invalido_debeRetornar422() throws Exception {
        String bodyInvalido = "{"nombre":"","email":"no-es-email"}";

        mockMvc.perform(post("/api/usuarios")
            .contentType(MediaType.APPLICATION_JSON)
            .content(bodyInvalido))
            .andExpect(status().isUnprocessableEntity())
            .andExpect(jsonPath("$.campos").exists());
    }
}
