6.1 Bean Validation

UsuarioRequest.java 
// UsuarioRequest.java — DTO de petición con validaciones
public record UsuarioRequest(
    @NotBlank(message = "El nombre es obligatorio")
    @Size(min = 2, max = 100, message = "Entre 2 y 100 caracteres")
    String nombre,

    @NotBlank(message = "El email es obligatorio")
    @Email(message = "Formato de email inválido")
    String email,

    @NotBlank(message = "La contraseña es obligatoria")
    @Size(min = 8, message = "Mínimo 8 caracteres")
    @Pattern(regexp = "^(?=.*[A-Z])(?=.*[0-9]).+$",
             message = "Debe tener al menos una mayúscula y un número")
    String password,

    @NotNull(message = "La edad es obligatoria")
    @Min(value = 18, message = "Debes ser mayor de edad")
    @Max(value = 120, message = "Edad inválida")
    Integer edad,

    @Past(message = "La fecha de nacimiento debe ser en el pasado")
    LocalDate fechaNacimiento,

    @Pattern(regexp = "^\d{9}$", message = "El teléfono debe tener 9 dígitos")
    String telefono,

    @URL(message = "La URL del perfil no es válida")
    String urlFoto
) {}

// Anotaciones de validación disponibles:
// @NotNull       — valor no puede ser null
// @NotBlank      — String no null, no vacío, no solo espacios
// @NotEmpty      — String/Collection no null y no vacío
// @Size          — longitud de String o tamaño de colección
// @Min / @Max    — valor numérico mínimo/máximo
// @Email         — formato email
// @Pattern       — expresión regular
// @Past / @Future — fechas pasadas/futuras
// @Positive / @Negative — números positivos/negativos
// @AssertTrue / @AssertFalse — booleano true/false

6.2 Manejador global de excepciones

GlobalExceptionHandler.java 
// GlobalExceptionHandler.java
@RestControllerAdvice  // intercepta excepciones de TODOS los controladores
public class GlobalExceptionHandler {

    // ── 404: recurso no encontrado ────────────────────────────
    @ExceptionHandler(RecursoNoEncontradoException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public Map<String, Object> handleNotFound(RecursoNoEncontradoException ex) {
        return Map.of(
            "error", "Recurso no encontrado",
            "mensaje", ex.getMessage(),
            "timestamp", LocalDateTime.now()
        );
    }

    // ── 400: errores de validación (@Valid) ───────────────────
    @ExceptionHandler(MethodArgumentNotValidException.class)
    @ResponseStatus(HttpStatus.UNPROCESSABLE_ENTITY)  // 422
    public Map<String, Object> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, String> errores = new HashMap<>();
        ex.getBindingResult().getFieldErrors().forEach(error ->
            errores.put(error.getField(), error.getDefaultMessage())
        );
        return Map.of(
            "error", "Errores de validación",
            "campos", errores,
            "timestamp", LocalDateTime.now()
        );
    }

    // ── 409: conflicto (email duplicado, etc.) ────────────────
    @ExceptionHandler(ConflictoException.class)
    @ResponseStatus(HttpStatus.CONFLICT)
    public Map<String, Object> handleConflicto(ConflictoException ex) {
        return Map.of("error", "Conflicto", "mensaje", ex.getMessage());
    }

    // ── 403 / 401: seguridad ──────────────────────────────────
    @ExceptionHandler(AccessDeniedException.class)
    @ResponseStatus(HttpStatus.FORBIDDEN)
    public Map<String, Object> handleForbidden(AccessDeniedException ex) {
        return Map.of("error", "Acceso denegado", "mensaje", ex.getMessage());
    }

    // ── 500: cualquier otra excepción ─────────────────────────
    @ExceptionHandler(Exception.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public Map<String, Object> handleGeneral(Exception ex) {
        // En producción: NO exponer detalles del error
        return Map.of(
            "error", "Error interno del servidor",
            "mensaje", "Contacte con el administrador"
        );
    }
}

// ── Excepciones personalizadas ────────────────────────────────
public class RecursoNoEncontradoException extends RuntimeException {
    public RecursoNoEncontradoException(String recurso, Long id) {
        super(recurso + " con id " + id + " no encontrado");
    }
}

public class ConflictoException extends RuntimeException {
    public ConflictoException(String mensaje) { super(mensaje); }
}
