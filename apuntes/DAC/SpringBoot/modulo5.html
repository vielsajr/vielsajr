5.1 Repositorios

UsuarioRepository.java 
// UsuarioRepository.java — Spring Data JPA genera la implementación automáticamente
@Repository
public interface UsuarioRepository extends JpaRepository<Usuario, Long> {

    // JpaRepository<Entidad, TipoPK> ya incluye:
    // save(), findById(), findAll(), delete(), count(), existsById()...

    // ── Métodos derivados del nombre ─────────────────────────
    // Spring genera el SQL automáticamente a partir del nombre del método
    Optional<Usuario> findByEmail(String email);
    List<Usuario> findByActivoTrue();
    List<Usuario> findByNombreContainingIgnoreCase(String nombre);
    List<Usuario> findByRolAndActivoTrue(Rol rol);
    boolean existsByEmail(String email);
    long countByActivoTrue();
    List<Usuario> findByNombreOrderByEmailAsc(String nombre);
    Optional<Usuario> findFirstByOrderByCreadoEnDesc();
    List<Usuario> findTop10ByActivoTrueOrderByNombreAsc();

    // ── @Query con JPQL (lenguaje de consulta sobre objetos) ──
    @Query("SELECT u FROM Usuario u WHERE u.email = :email AND u.activo = true")
    Optional<Usuario> findActivoByEmail(@Param("email") String email);

    @Query("SELECT u FROM Usuario u WHERE LOWER(u.nombre) LIKE LOWER(CONCAT('%', :termino, '%'))")
    List<Usuario> buscarPorNombre(@Param("termino") String termino);

    // ── @Query con SQL nativo ─────────────────────────────────
    @Query(value = "SELECT * FROM usuarios WHERE DATE(creado_en) = CURDATE()",
           nativeQuery = true)
    List<Usuario> findCreadosHoy();

    // ── @Modifying para UPDATE / DELETE ──────────────────────
    @Modifying
    @Transactional
    @Query("UPDATE Usuario u SET u.activo = false WHERE u.id = :id")
    void desactivar(@Param("id") Long id);

    @Modifying
    @Transactional
    @Query("DELETE FROM Usuario u WHERE u.activo = false AND u.creadoEn < :fecha")
    int eliminarInactivosAnterioresA(@Param("fecha") LocalDateTime fecha);

    // ── Paginación ────────────────────────────────────────────
    Page<Usuario> findByActivoTrue(Pageable pageable);
    Slice<Usuario> findByRol(Rol rol, Pageable pageable);
}

5.2 Capa de servicio con @Transactional

UsuarioServiceImpl.java 
// UsuarioServiceImpl.java — Implementación de la lógica de negocio
@Service
@RequiredArgsConstructor
public class UsuarioServiceImpl implements UsuarioService {

    private final UsuarioRepository repo;
    private final PasswordEncoder passwordEncoder;
    private final UsuarioMapper mapper;  // MapStruct o manual

    @Override
    @Transactional(readOnly = true)  // optimización para consultas
    public Page<UsuarioDTO> listarTodos(Pageable pageable) {
        return repo.findAll(pageable).map(mapper::toDTO);
    }

    @Override
    @Transactional(readOnly = true)
    public UsuarioDTO obtenerPorId(Long id) {
        return repo.findById(id)
            .map(mapper::toDTO)
            .orElseThrow(() -> new RecursoNoEncontradoException("Usuario", id));
    }

    @Override
    @Transactional  // apertura de transacción por defecto
    public UsuarioDTO crear(UsuarioRequest request) {
        if (repo.existsByEmail(request.email())) {
            throw new ConflictoException("Ya existe un usuario con ese email");
        }
        Usuario usuario = Usuario.builder()
            .nombre(request.nombre())
            .email(request.email())
            .password(passwordEncoder.encode(request.password()))
            .rol(Rol.USUARIO)
            .activo(true)
            .build();
        return mapper.toDTO(repo.save(usuario));
    }

    @Override
    @Transactional
    public UsuarioDTO actualizar(Long id, UsuarioRequest request) {
        Usuario usuario = repo.findById(id)
            .orElseThrow(() -> new RecursoNoEncontradoException("Usuario", id));
        usuario.setNombre(request.nombre());
        usuario.setEmail(request.email());
        if (request.password() != null && !request.password().isBlank()) {
            usuario.setPassword(passwordEncoder.encode(request.password()));
        }
        return mapper.toDTO(repo.save(usuario));  // save() hace UPDATE si el id existe
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        if (!repo.existsById(id)) {
            throw new RecursoNoEncontradoException("Usuario", id);
        }
        repo.deleteById(id);
    }
}
@Transactional(readOnly=true) en consultas mejora el rendimiento. @Transactional sin parámetros abre una transacción que se hace commit al terminar o rollback si hay excepción.
