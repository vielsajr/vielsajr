3.1 Controlador REST completo

UsuarioController.java 
// UsuarioController.java — REST API completa
@RestController
@RequestMapping("/api/usuarios")
@RequiredArgsConstructor
public class UsuarioController {

    private final UsuarioService usuarioService;

    // ── GET /api/usuarios ─────────────────────────────────────
    @GetMapping
    public ResponseEntity<Page<UsuarioDTO>> listar(
            @RequestParam(defaultValue="0") int page,
            @RequestParam(defaultValue="20") int size,
            @RequestParam(defaultValue="nombre") String sort) {
        Pageable pageable = PageRequest.of(page, size, Sort.by(sort));
        return ResponseEntity.ok(usuarioService.listarTodos(pageable));
    }

    // ── GET /api/usuarios/1 ───────────────────────────────────
    @GetMapping("/{id}")
    public ResponseEntity<UsuarioDTO> obtener(@PathVariable Long id) {
        return ResponseEntity.ok(usuarioService.obtenerPorId(id));
    }

    // ── POST /api/usuarios ────────────────────────────────────
    @PostMapping
    public ResponseEntity<UsuarioDTO> crear(
            @Valid @RequestBody UsuarioRequest request) {
        UsuarioDTO creado = usuarioService.crear(request);
        URI location = ServletUriComponentsBuilder.fromCurrentRequest()
            .path("/{id}").buildAndExpand(creado.id()).toUri();
        return ResponseEntity.created(location).body(creado);
    }

    // ── PUT /api/usuarios/1 ───────────────────────────────────
    @PutMapping("/{id}")
    public ResponseEntity<UsuarioDTO> actualizar(
            @PathVariable Long id,
            @Valid @RequestBody UsuarioRequest request) {
        return ResponseEntity.ok(usuarioService.actualizar(id, request));
    }

    // ── PATCH /api/usuarios/1 ────────────────────────────────
    @PatchMapping("/{id}")
    public ResponseEntity<UsuarioDTO> patchear(
            @PathVariable Long id,
            @RequestBody Map<String, Object> cambios) {
        return ResponseEntity.ok(usuarioService.patchear(id, cambios));
    }

    // ── DELETE /api/usuarios/1 ───────────────────────────────
    @DeleteMapping("/{id}")
    public ResponseEntity<Void> eliminar(@PathVariable Long id) {
        usuarioService.eliminar(id);
        return ResponseEntity.noContent().build();  // 204
    }

    // ── GET /api/usuarios/buscar?nombre=ana&email=... ──────────
    @GetMapping("/buscar")
    public ResponseEntity<List<UsuarioDTO>> buscar(
            @RequestParam(required=false) String nombre,
            @RequestParam(required=false) String email) {
        return ResponseEntity.ok(usuarioService.buscar(nombre, email));
    }
}

3.2 Anotaciones de mapeo y parámetros

AnotaciónDescripción
@GetMapping("/ruta")Mapea peticiones GET
@PostMapping("/ruta")Mapea peticiones POST
@PutMapping("/ruta")Mapea peticiones PUT
@PatchMapping("/ruta")Mapea peticiones PATCH
@DeleteMapping("/ruta")Mapea peticiones DELETE
@RequestMapping("/base")Prefijo de ruta para toda la clase
@PathVariableExtrae variable de la URL: /usuarios/{id}
@RequestParamExtrae parámetro de query string: ?page=2
@RequestBodyDeserializa el body JSON a objeto Java
@RequestHeaderExtrae cabecera HTTP: @RequestHeader("Authorization")
@CookieValueExtrae valor de una cookie
@Valid / @ValidatedActiva la validación del objeto anotado (Bean Validation)
ResponseEntityPermite controlar código de estado, cabeceras y cuerpo
@ResponseStatus(HttpStatus.CREATED)Establece el código HTTP de la respuesta
@CrossOriginHabilita CORS para ese controlador o método

3.3 ResponseEntity y códigos HTTP

ResponseEntityEjemplos.java 
// ResponseEntity: control total sobre la respuesta HTTP

// 200 OK con cuerpo
return ResponseEntity.ok(datos);

// 201 Created con Location header
return ResponseEntity.created(uri).body(creado);

// 204 No Content (para DELETE exitoso)
return ResponseEntity.noContent().build();

// 400 Bad Request
return ResponseEntity.badRequest().body(Map.of("error", "Datos inválidos"));

// 404 Not Found
return ResponseEntity.notFound().build();

// Personalizado con código arbitrario
return ResponseEntity.status(HttpStatus.ACCEPTED).body(datos);
return ResponseEntity.status(429).body(Map.of("error", "Too many requests"));

// Con cabeceras personalizadas
HttpHeaders headers = new HttpHeaders();
headers.add("X-Total-Count", String.valueOf(total));
return ResponseEntity.ok().headers(headers).body(lista);
