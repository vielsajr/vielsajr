3.1 Anatomía de un componente

usuario-card.component.ts
// usuario-card.component.ts — Componente completo
import { Component, Input, Output, EventEmitter,
         OnInit, OnChanges, OnDestroy, SimpleChanges,
         ChangeDetectionStrategy } from '@angular/core';
import { CommonModule } from '@angular/common';
import { RouterModule } from '@angular/router';
import { Usuario } from '../../modelos/usuario.model';

@Component({
  selector: 'app-usuario-card',  // <app-usuario-card> en el HTML
  standalone: true,
  imports: [CommonModule, RouterModule],  // módulos que usa la plantilla
  templateUrl: './usuario-card.component.html',
  styleUrls: ['./usuario-card.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush,  // optimización
})
export class UsuarioCardComponent implements OnInit, OnChanges, OnDestroy {

  // ── @Input: datos que recibe del componente padre ────────
  @Input({ required: true }) usuario!: Usuario;  // ! → no null assertion
  @Input() mostrarDetalle = false;
  @Input() set color(valor: string) {  // setter con lógica
    this._color = valor.toLowerCase();
  }
  private _color = 'blue';

  // ── @Output: eventos que emite al componente padre ───────
  @Output() seleccionado = new EventEmitter<Usuario>();
  @Output() eliminado = new EventEmitter<number>();  // emite el id

  // ── Propiedades internas ──────────────────────────────────
  protected esFavorito = false;
  protected cargando = false;

  // ── Ciclo de vida ─────────────────────────────────────────
  ngOnChanges(changes: SimpleChanges): void {
    if (changes['usuario']) {
      console.log('Usuario cambió:', changes['usuario'].currentValue);
    }
  }

  ngOnInit(): void {
    console.log('Componente inicializado para:', this.usuario.nombre);
  }

  ngOnDestroy(): void {
    // Desuscribir Observables aquí para evitar memory leaks
  }

  // ── Métodos del componente ────────────────────────────────
  onSeleccionar(): void {
    this.seleccionado.emit(this.usuario);
  }

  onEliminar(event: Event): void {
    event.stopPropagation();  // evitar que se propague al padre
    this.eliminado.emit(this.usuario.id);
  }

  toggleFavorito(): void {
    this.esFavorito = !this.esFavorito;
  }
}

3.2 Ciclo de vida completo

HookCuándo se llama
ngOnChangesCuando cambia un @Input. Recibe SimpleChanges con anterior y nuevo valor. SE LLAMA ANTES de ngOnInit.
ngOnInitUna vez, tras el primer ngOnChanges. Inicialización del componente. Cargar datos de APIs aquí.
ngDoCheckCada ciclo de detección de cambios. Usar con precaución (costoso en rendimiento).
ngAfterContentInitDespués de proyectar contenido con ng-content. Una sola vez.
ngAfterContentCheckedDespués de cada comprobación del contenido proyectado.
ngAfterViewInitDespués de inicializar la vista y las vistas hijas. Para acceder a @ViewChild.
ngAfterViewCheckedDespués de cada comprobación de la vista y sus hijas.
ngOnDestroyJusto antes de destruir el componente. Desuscribir Observables, limpiar timers.

3.3 Comunicación entre componentes

comunicacion-componentes.ts
// ── Padre → Hijo: @Input ───────────────────────────────────────
// En el padre (template):
// <app-usuario-card [usuario]="usuarioSeleccionado" [mostrarDetalle]="true">
// </app-usuario-card>

// ── Hijo → Padre: @Output + EventEmitter ──────────────────────
// En el padre (template):
// <app-usuario-card
//   [usuario]="usuario"
//   (seleccionado)="onUsuarioSeleccionado($event)"
//   (eliminado)="onEliminar($event)">
// </app-usuario-card>

// En el padre (TypeScript):
onUsuarioSeleccionado(usuario: Usuario): void {
  this.usuarioActual = usuario;
}
onEliminar(id: number): void {
  this.usuarios = this.usuarios.filter(u => u.id !== id);
}

// ── Padre accede a hijo: @ViewChild ───────────────────────────
@Component({...})
export class PadreComponent implements AfterViewInit {
  @ViewChild(HijoComponent) hijoRef!: HijoComponent;
  @ViewChild('miInput') inputRef!: ElementRef<HTMLInputElement>;

  ngAfterViewInit(): void {
    this.hijoRef.metodoPublico();
    this.inputRef.nativeElement.focus();
  }
}
