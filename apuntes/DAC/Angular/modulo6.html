6.1 Servicios (@Injectable)

usuario.service.ts
// usuario.service.ts — Servicio completo
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable, BehaviorSubject, throwError } from 'rxjs';
import { tap, catchError, map, shareReplay } from 'rxjs/operators';
import { environment } from '../../environments/environment';
import { Usuario, UsuarioRequest, Pagina } from '../modelos/usuario.model';

@Injectable({ providedIn: 'root' })  // singleton en toda la app
export class UsuarioService {
  private http = inject(HttpClient);  // nueva forma sin constructor
  private api = `${environment.apiUrl}/usuarios`;

  // ── Estado local del servicio ─────────────────────────────
  private usuariosSubject = new BehaviorSubject<Usuario[]>([]);
  usuarios$ = this.usuariosSubject.asObservable();  // solo lectura

  // ── CRUD con HttpClient ───────────────────────────────────
  getAll(page = 0, size = 20): Observable<Pagina<Usuario>> {
    return this.http.get<Pagina<Usuario>>(`${this.api}?page=${page}&size=${size}`);
  }

  getById(id: number): Observable<Usuario> {
    return this.http.get<Usuario>(`${this.api}/${id}`).pipe(
      catchError(err => throwError(() =>
        new Error(err.error?.mensaje || `Error ${err.status}`)))
    );
  }

  create(datos: UsuarioRequest): Observable<Usuario> {
    return this.http.post<Usuario>(this.api, datos).pipe(
      tap(nuevo => {
        const actuales = this.usuariosSubject.getValue();
        this.usuariosSubject.next([...actuales, nuevo]);
      })
    );
  }

  update(id: number, datos: Partial<UsuarioRequest>): Observable<Usuario> {
    return this.http.patch<Usuario>(`${this.api}/${id}`, datos);
  }

  delete(id: number): Observable<void> {
    return this.http.delete<void>(`${this.api}/${id}`).pipe(
      tap(() => {
        const actuales = this.usuariosSubject.getValue();
        this.usuariosSubject.next(actuales.filter(u => u.id !== id));
      })
    );
  }

  // ── Búsqueda con caché ────────────────────────────────────
  private cache$?: Observable<Usuario[]>;
  getAllCacheado(): Observable<Usuario[]> {
    if (!this.cache$) {
      this.cache$ = this.http.get<Usuario[]>(this.api).pipe(
        shareReplay(1)  // cachea el último resultado
      );
    }
    return this.cache$;
  }
}

6.2 Operadores RxJS esenciales

OperadorDescripción
map(fn)Transforma cada valor emitido: .pipe(map(u => u.nombre))
filter(pred)Filtra valores: .pipe(filter(u => u.activo))
tap(fn)Efecto secundario sin transformar: logging, actualizar estado
catchError(fn)Captura errores y devuelve un Observable alternativo o throwError
switchMap(fn)Cancela la petición anterior y hace la nueva. Ideal para búsquedas.
mergeMap(fn)Ejecuta en paralelo todas las peticiones inner.
concatMap(fn)Ejecuta en serie (una tras otra, en orden).
exhaustMap(fn)Ignora nuevas peticiones mientras hay una en curso. Ideal para login.
debounceTime(ms)Espera N ms de silencio antes de emitir. Para búsquedas en vivo.
distinctUntilChanged()Solo emite si el valor cambió respecto al anterior.
takeUntil(obs$)Se desuscribe cuando obs$ emite. Para limpiar en ngOnDestroy.
take(n)Toma solo los primeros N valores y completa.
shareReplay(1)Comparte el Observable y cachea el último resultado.
combineLatest([])Combina múltiples Observables, emite cuando cualquiera cambia.
forkJoin([])Como Promise.all: emite cuando TODOS han completado.
of(valor)Crea Observable que emite el valor y completa.
from(promesa)Convierte Promise en Observable.
EMPTYObservable que completa inmediatamente sin emitir.

6.3 Patrones RxJS en Angular

rxjs-patrones.ts
// ── Patrones RxJS más usados en Angular ───────────────────────

// 1. Búsqueda en tiempo real con debounce
this.searchControl.valueChanges.pipe(
  debounceTime(300),
  distinctUntilChanged(),
  filter(term => term.length >= 2),
  switchMap(term => this.service.buscar(term).pipe(
    catchError(() => of([]))  // error → lista vacía
  ))
).subscribe(resultados => this.resultados = resultados);

// 2. Desuscripción limpia con takeUntil + Subject
@Component({...})
export class MiComponente implements OnDestroy {
  private destroy$ = new Subject<void>();

  ngOnInit(): void {
    this.service.datos$.pipe(
      takeUntil(this.destroy$)  // se desuscribe cuando el componente muere
    ).subscribe(datos => this.datos = datos);
  }

  ngOnDestroy(): void {
    this.destroy$.next();
    this.destroy$.complete();
  }
}

// 3. Combinar múltiples peticiones
combineLatest([
  this.usuarioService.getById(id),
  this.pedidoService.getByUsuario(id),
]).pipe(
  map(([usuario, pedidos]) => ({ usuario, pedidos }))
).subscribe(datos => this.datosCompletos = datos);

// 4. Subject como bus de eventos
// En el servicio:
private notificacionSubject = new Subject<string>();
notificacion$ = this.notificacionSubject.asObservable();
emitir(msg: string): void { this.notificacionSubject.next(msg); }

// En el componente:
this.notifService.notificacion$
  .pipe(takeUntil(this.destroy$))
  .subscribe(msg => this.mostrarToast(msg));
