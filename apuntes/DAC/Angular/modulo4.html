MÓDULO 04

Template Syntax: Directivas y Data Binding

Los cuatro tipos de data binding, las nuevas directivas de control de flujo @if/@for/@switch (Angular 17+) y directivas de atributo.

Contenido
4.1 Data Binding4.2 Directivas estructurales4.3 Directivas de atributo

4.1 Data Binding

data-binding.html
<!-- ── Interpolación: componente → DOM ────────────────────────── -->
<h1>{{ titulo }}</h1>
<p>{{ usuario.nombre | uppercase }}</p>
<span>{{ 1 + 1 }}</span>
<span>{{ estaActivo ? 'Activo' : 'Inactivo' }}</span>

<!-- ── Property Binding: componente → DOM ──────────────────────── -->
<img [src]="usuario.foto" [alt]="usuario.nombre">
<button [disabled]="cargando">Enviar</button>
<div [class.activo]="estaSeleccionado">...</div>
<div [style.color]="colorTexto">...</div>
<input [value]="nombre">

<!-- ── Event Binding: DOM → componente ─────────────────────────── -->
<button (click)="guardar()">Guardar</button>
<input (input)="onCambio($event)">
<input (keyup.enter)="buscar()">
<form (ngSubmit)="onSubmit()">
<div (mouseenter)="onHover()" (mouseleave)="onHoverOut()">

<!-- ── Two-way Binding: [(ngModel)] ─────────────────────────────── -->
<!-- Requiere importar FormsModule -->
<input [(ngModel)]="nombre" placeholder="Nombre">
<!-- Equivale a: -->
<input [value]="nombre" (input)="nombre = $event.target.value">

<!-- ── Template Reference Variable ──────────────────────────────── -->
<input #miInput type="text">
<button (click)="miInput.focus()">Enfocar</button>
<p>Valor: {{ miInput.value }}</p>

4.2 Directivas estructurales

directivas-estructurales.html
<!-- ── @if / @else (Angular 17+ — nuevo control flow) ──────────── -->
@if (usuario) {
  <div class="perfil">{{ usuario.nombre }}</div>
} @else if (cargando) {
  <app-spinner></app-spinner>
} @else {
  <p>No se encontró el usuario</p>
}

<!-- ── Forma antigua con *ngIf (sigue funcionando) ──────────────── -->
<div *ngIf="usuario; else noHayUsuario">{{ usuario.nombre }}</div>
<ng-template #noHayUsuario><p>No hay usuario</p></ng-template>

<!-- ── @for (Angular 17+ — nuevo control flow) ──────────────────── -->
@for (item of lista; track item.id) {
  <div class="item">
    <span>{{ $index + 1 }}.</span>
    <strong>{{ item.nombre }}</strong>
    @if ($first) { <span class="badge">Primero</span> }
    @if ($last) { <span class="badge">Último</span> }
  </div>
} @empty {
  <p>La lista está vacía</p>
}

<!-- ── Forma antigua con *ngFor ───────────────────────────────────── -->
<li *ngFor="let item of lista; let i = index; trackBy: trackById">
  {{ i }} — {{ item.nombre }}
</li>
// En el .ts para trackBy (mejora rendimiento):
trackById(index: number, item: any): number { return item.id; }

<!-- ── @switch (Angular 17+) ────────────────────────────────────── -->
@switch (estado) {
  @case ('activo')   { <span class="verde">Activo</span> }
  @case ('inactivo') { <span class="rojo">Inactivo</span> }
  @default           { <span>Desconocido</span> }
}

4.3 Directivas de atributo

directivas-atributo.html
<!-- ── ngClass: clases CSS dinámicas ────────────────────────────── -->
<div [ngClass]="{'activo': estaActivo, 'resaltado': estaResaltado}">...</div>
<div [ngClass]="obtenerClases()">...</div>
<div [class]="'btn ' + tipoBton">...</div>

<!-- ── ngStyle: estilos inline dinámicos ───────────────────────── -->
<div [ngStyle]="{'color': colorTexto, 'font-size': tamanio + 'px'}">...</div>

<!-- ── Directiva personalizada de atributo ─────────────────────── -->
// highlight.directive.ts
@Directive({
  selector: '[appHighlight]',
  standalone: true,
})
export class HighlightDirective {
  @Input('appHighlight') colorFondo = 'yellow';

  @HostListener('mouseenter')
  onMouseEnter(): void {
    this.el.nativeElement.style.backgroundColor = this.colorFondo;
  }

  @HostListener('mouseleave')
  onMouseLeave(): void {
    this.el.nativeElement.style.backgroundColor = '';
  }

  constructor(private el: ElementRef) {}
}

<!-- Uso en template: -->
<!-- <p appHighlight="lightblue">Texto resaltado</p> -->
