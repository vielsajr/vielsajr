8.1 HttpClient completo

http-client.ts
// ── HttpClient — todos los métodos ────────────────────────────
private http = inject(HttpClient);

// GET con parámetros
this.http.get<Usuario[]>('/api/usuarios', {
  params: new HttpParams()
    .set('page', '0').set('size', '20').set('sort', 'nombre'),
  headers: new HttpHeaders({ 'X-Custom': 'valor' }),
  observe: 'response',  // obtener el HttpResponse completo
}).subscribe(res => {
  console.log('Status:', res.status);
  console.log('Datos:', res.body);
  console.log('Total:', res.headers.get('X-Total-Count'));
});

// POST con JSON
this.http.post<Usuario>('/api/usuarios', { nombre: 'Ana', email: 'ana@mail.com' })
  .subscribe(nuevo => console.log('Creado:', nuevo));

// PUT / PATCH
this.http.put<Usuario>('/api/usuarios/1', datos).subscribe(...);
this.http.patch<Usuario>('/api/usuarios/1', { nombre: 'Ana M.' }).subscribe(...);

// DELETE
this.http.delete<void>('/api/usuarios/1').subscribe(...);

// Subir archivo con FormData + progreso
const fd = new FormData();
fd.append('archivo', file);
fd.append('nombre', 'foto-perfil');

this.http.post('/api/uploads', fd, {
  reportProgress: true,
  observe: 'events',
}).pipe(
  filter(event => event.type === HttpEventType.UploadProgress),
  map(event => Math.round(100 * (event as any).loaded / (event as any).total))
).subscribe(pct => this.progreso = pct);

8.2 Interceptors

jwt.interceptor.ts
// jwt.interceptor.ts — Añade token JWT a todas las peticiones
import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { AuthService } from '../servicios/auth.service';
import { catchError, throwError } from 'rxjs';
import { Router } from '@angular/router';

export const jwtInterceptor: HttpInterceptorFn = (req, next) => {
  const auth = inject(AuthService);
  const router = inject(Router);
  const token = auth.getToken();

  // Clonar la petición y añadir cabecera (las peticiones son inmutables)
  const reqConToken = token
    ? req.clone({ setHeaders: { Authorization: `Bearer ${token}` } })
    : req;

  return next(reqConToken).pipe(
    catchError(error => {
      if (error.status === 401) {
        auth.logout();
        router.navigate(['/login']);
      }
      return throwError(() => error);
    })
  );
};

// loading.interceptor.ts — Spinner global
export const loadingInterceptor: HttpInterceptorFn = (req, next) => {
  const loadingService = inject(LoadingService);
  loadingService.mostrar();
  return next(req).pipe(
    finalize(() => loadingService.ocultar())
  );
};
