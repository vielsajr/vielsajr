1.1 Tipos básicos, interfaces y genéricos

TypeScript es JavaScript con tipos estáticos. Angular está completamente escrito en TypeScript y todos sus proyectos lo usan por defecto.
typescript-basico.ts
// ── Tipos básicos ──────────────────────────────────────────────
let nombre: string = 'Ana';
let edad: number = 28;
let activo: boolean = true;
let lista: string[] = ['a', 'b', 'c'];
let tupla: [string, number] = ['Ana', 28];
let cualquier: any = 'puede ser cualquier cosa';  // evitar
let desconocido: unknown = 42;  // más seguro que any
let vacio: void;   // para funciones que no retornan nada
let nulo: null = null;
let undef: undefined = undefined;

// ── Union types y type aliases ──────────────────────────────────
type Estado = 'activo' | 'inactivo' | 'pendiente';  // literal union
type Id = string | number;  // union de tipos
type Nullable<T> = T | null;  // genérico

// ── Interfaces (para objetos) ───────────────────────────────────
interface Usuario {
  readonly id: number;  // readonly: no se puede cambiar
  nombre: string;
  email: string;
  edad?: number;      // ?: propiedad opcional
  telefono?: string;
}

// ── Enums ──────────────────────────────────────────────────────
enum Rol { ADMIN = 'ADMIN', USUARIO = 'USUARIO', MODERADOR = 'MODERADOR' }

// ── Genéricos ──────────────────────────────────────────────────
interface ApiResponse<T> {
  data: T;
  mensaje: string;
  ok: boolean;
}
// Uso: ApiResponse<Usuario>, ApiResponse<Producto[]>

// ── Utility types más usados en Angular ───────────────────────
type UsuarioParcial  = Partial<Usuario>;         // todas las props opcionales
type UsuarioReq      = Required<Usuario>;        // todas las props obligatorias
type UsuarioSoloId   = Pick<Usuario, 'id' | 'nombre'>;  // solo esas props
type UsuarioSinEmail = Omit<Usuario, 'email'>;   // todas menos email
type IdMap           = Record<string, Usuario>;  // { [key: string]: Usuario }

1.2 Clases y decoradores

clases-decoradores.ts
// ── Clases en TypeScript ──────────────────────────────────────
class Animal {
  // Shorthand: declara Y asigna en el constructor
  constructor(
    public nombre: string,
    protected tipo: string,
    private edad: number
  ) {}

  hablar(): string { return `${this.nombre} dice algo`; }
}

class Perro extends Animal {
  constructor(nombre: string, private raza: string) {
    super(nombre, 'mamifero', 3);
  }
  override hablar(): string { return `${this.nombre} dice Guau`; }
  get descripcion(): string { return `${this.nombre} (${this.raza})`; }
}

// ── Decoradores (fundamentales en Angular) ────────────────────
// Un decorador es una función que modifica una clase, método o propiedad

// Decorador de clase:
function Loggable(target: Function) {
  console.log('Clase creada:', target.name);
}

@Loggable
class MiServicio { }

// En Angular, los decoradores más importantes son:
// @Component, @Injectable, @NgModule, @Pipe, @Directive
// @Input, @Output, @ViewChild, @HostListener, @HostBinding
