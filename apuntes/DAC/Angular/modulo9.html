Formularios: Template-Driven y Reactive

Template-Driven Forms con ngModel, Reactive Forms completos con FormBuilder, FormGroup, FormArray y validadores personalizados de grupo.

9.1 Template-Driven Forms

Los formularios template-driven usan directivas en el HTML. Son más simples pero menos flexibles. Requieren importar FormsModule.
template-driven.html
<!-- registro.component.html — Template-Driven Form -->
<form #miForm="ngForm" (ngSubmit)="onSubmit(miForm)">
  <div class="campo">
    <label for="nombre">Nombre</label>
    <input id="nombre" name="nombre"
           [(ngModel)]="datos.nombre"
           #nombreCtrl="ngModel"
           required minlength="2" maxlength="100">
    <div *ngIf="nombreCtrl.invalid && nombreCtrl.touched">
      <span *ngIf="nombreCtrl.errors?.['required']">El nombre es obligatorio</span>
      <span *ngIf="nombreCtrl.errors?.['minlength']">Mínimo 2 caracteres</span>
    </div>
  </div>

  <div class="campo">
    <label for="email">Email</label>
    <input id="email" name="email" type="email"
           [(ngModel)]="datos.email"
           #emailCtrl="ngModel"
           required email>
    <div *ngIf="emailCtrl.invalid && emailCtrl.touched">
      <span *ngIf="emailCtrl.errors?.['email']">Email inválido</span>
    </div>
  </div>

  <button type="submit" [disabled]="miForm.invalid || cargando">
    {{ cargando ? 'Enviando...' : 'Registrarse' }}
  </button>
</form>
template-driven.ts
// registro.component.ts — Template-Driven
@Component({ imports: [FormsModule, CommonModule], ... })
export class RegistroComponent {
  datos = { nombre: '', email: '', password: '' };
  cargando = false;

  onSubmit(form: NgForm): void {
    if (form.invalid) return;
    this.cargando = true;
    this.service.crear(this.datos).subscribe({
      next: () => { this.router.navigate(['/exito']); },
      error: (err) => { this.error = err.message; this.cargando = false; },
    });
  }
}

9.2 Reactive Forms

Los Reactive Forms definen la estructura del formulario en TypeScript. Son más verbosos pero ofrecen mucho más control, validación dinámica y se testean mejor.
reactive-form.ts
// formulario.component.ts — Reactive Form completo
@Component({ imports: [ReactiveFormsModule, CommonModule], ... })
export class FormularioComponent implements OnInit {
  private fb = inject(FormBuilder);
  formulario!: FormGroup;

  ngOnInit(): void {
    this.formulario = this.fb.group({
      nombre: ['', [Validators.required, Validators.minLength(2), Validators.maxLength(100)]],
      email: ['', [Validators.required, Validators.email]],
      password: ['', [
        Validators.required,
        Validators.minLength(8),
        Validators.pattern(/^(?=.*[A-Z])(?=.*\d).+$/),  // regex
      ]],
      confirmar: ['', Validators.required],
      edad: [null, [Validators.required, Validators.min(18), Validators.max(120)]],
      direccion: this.fb.group({  // FormGroup anidado
        calle: ['', Validators.required],
        ciudad: ['', Validators.required],
        cp: ['', Validators.pattern(/^\d{5}$/)],
      }),
      intereses: this.fb.array([]),  // FormArray dinámico
    }, {
      validators: this.passwordsIgualesValidator,  // validator del grupo
    });
  }

  // ── Acceder a controles ───────────────────────────────────
  get f() { return this.formulario.controls; }
  get nombreCtrl() { return this.formulario.get('nombre')!; }
  get interesesArray() { return this.formulario.get('intereses') as FormArray; }

  // ── FormArray: agregar/eliminar dinámicamente ─────────────
  agregarInteres(valor: string): void {
    this.interesesArray.push(this.fb.control(valor, Validators.required));
  }
  eliminarInteres(i: number): void { this.interesesArray.removeAt(i); }

  // ── Validator personalizado del grupo ─────────────────────
  passwordsIgualesValidator(group: AbstractControl): ValidationErrors | null {
    const pass = group.get('password')?.value;
    const confirm = group.get('confirmar')?.value;
    return pass === confirm ? null : { noCoinciden: true };
  }

  // ── Parchear valores ──────────────────────────────────────
  cargarDatos(usuario: Usuario): void {
    this.formulario.patchValue({  // solo actualiza los campos indicados
      nombre: usuario.nombre,
      email: usuario.email,
    });
    // setValue requiere TODOS los campos del formulario
  }

  // ── Submit ────────────────────────────────────────────────
  onSubmit(): void {
    if (this.formulario.invalid) {
      this.formulario.markAllAsTouched();  // mostrar todos los errores
      return;
    }
    const datos = this.formulario.value;
    this.service.crear(datos).subscribe({ next: () => {}, error: () => {} });
  }
}
reactive-form.html
<!-- reactive-form.html — Template para Reactive Form -->
<form [formGroup]="formulario" (ngSubmit)="onSubmit()">
  <div class="campo">
    <label>Nombre</label>
    <input formControlName="nombre">
    <div *ngIf="f['nombre'].invalid && f['nombre'].touched">
      <span *ngIf="f['nombre'].errors?.['required']">Obligatorio</span>
      <span *ngIf="f['nombre'].errors?.['minlength']">
        Mínimo {{ f['nombre'].errors?.['minlength'].requiredLength }} caracteres
      </span>
    </div>
  </div>

  <!-- FormGroup anidado -->
  <div formGroupName="direccion">
    <input formControlName="calle" placeholder="Calle">
    <input formControlName="ciudad" placeholder="Ciudad">
  </div>

  <!-- FormArray -->
  <div formArrayName="intereses">
    <div *ngFor="let ctrl of interesesArray.controls; let i = index">
      <input [formControlName]="i">
      <button type="button" (click)="eliminarInteres(i)">X</button>
    </div>
  </div>

  <!-- Error de grupo (contraseñas no coinciden) -->
  <span *ngIf="formulario.errors?.['noCoinciden']">Las contraseñas no coinciden</span>

  <button type="submit" [disabled]="formulario.invalid">Enviar</button>
</form>
