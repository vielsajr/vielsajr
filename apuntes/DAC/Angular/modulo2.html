2.1 Instalación y CLI

CLI de Angular
# ── Instalación ──────────────────────────────────────────────
npm install -g @angular/cli
ng version

# ── Crear proyecto ───────────────────────────────────────────
ng new mi-app
# ¿Habilitar routing? → Sí
# ¿Formato de estilos? → CSS / SCSS
cd mi-app
ng serve            # servidor de desarrollo en http://localhost:4200
ng serve --open     # abre el navegador automáticamente
ng build            # build de producción en dist/
ng build --prod     # con optimizaciones (minify, tree-shaking, AOT)

# ── Generadores (schematics) ──────────────────────────────────
ng generate component usuarios/lista      # o: ng g c usuarios/lista
ng generate service servicios/usuario     # ng g s servicios/usuario
ng generate module usuarios --routing     # ng g m usuarios --routing
ng generate pipe pipes/truncar            # ng g p pipes/truncar
ng generate directive directivas/highlight # ng g d directivas/highlight
ng generate guard guards/auth             # ng g guard guards/auth
ng generate interceptor interceptores/jwt # ng g interceptor ...
ng generate interface modelos/usuario     # ng g i modelos/usuario
ng generate enum modelos/rol              # ng g enum modelos/rol

# ── Testing ───────────────────────────────────────────────────
ng test   # lanza Karma + Jasmine
ng e2e    # tests end-to-end

2.3 Componente standalone (Angular 17+)

Desde Angular 17, los componentes son standalone por defecto: no necesitan NgModule. Esto simplifica mucho la arquitectura.
app.config.ts
// app.config.ts — Configuración principal (standalone)
import { ApplicationConfig } from '@angular/core';
import { provideRouter } from '@angular/router';
import { provideHttpClient, withInterceptors } from '@angular/common/http';
import { routes } from './app.routes';
import { jwtInterceptor } from './interceptores/jwt.interceptor';

export const appConfig: ApplicationConfig = {
  providers: [
    provideRouter(routes),
    provideHttpClient(withInterceptors([jwtInterceptor])),
  ]
};

// main.ts
import { bootstrapApplication } from '@angular/platform-browser';
import { appConfig } from './app/app.config';
import { AppComponent } from './app/app.component';

bootstrapApplication(AppComponent, appConfig)
  .catch(err => console.error(err));
