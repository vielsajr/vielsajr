12.1 Tests de servicios

usuario.service.spec.ts
// usuario.service.spec.ts
import { TestBed } from '@angular/core/testing';
import { HttpClientTestingModule, HttpTestingController } from '@angular/common/http/testing';
import { UsuarioService } from './usuario.service';

describe('UsuarioService', () => {
  let service: UsuarioService;
  let httpMock: HttpTestingController;

  beforeEach(() => {
    TestBed.configureTestingModule({
      imports: [HttpClientTestingModule],
      providers: [UsuarioService],
    });
    service = TestBed.inject(UsuarioService);
    httpMock = TestBed.inject(HttpTestingController);
  });

  afterEach(() => httpMock.verify());  // verificar no hay peticiones pendientes

  it('debe obtener lista de usuarios', () => {
    const mockUsuarios = [{ id: 1, nombre: 'Ana' }, { id: 2, nombre: 'Luis' }];

    service.getAll().subscribe(lista => {
      expect(lista.length).toBe(2);
      expect(lista[0].nombre).toBe('Ana');
    });

    const req = httpMock.expectOne('/api/usuarios?page=0&size=20');
    expect(req.request.method).toBe('GET');
    req.flush(mockUsuarios);  // simular respuesta del servidor
  });

  it('debe lanzar error cuando el servidor devuelve 404', () => {
    service.getById(999).subscribe({
      error: (err) => expect(err.message).toContain('404'),
    });

    const req = httpMock.expectOne('/api/usuarios/999');
    req.flush('Not Found', { status: 404, statusText: 'Not Found' });
  });
});

12.2 Tests de componentes con TestBed

usuario-lista.component.spec.ts
// usuario-lista.component.spec.ts
import { ComponentFixture, TestBed } from '@angular/core/testing';
import { By } from '@angular/platform-browser';
import { of, throwError } from 'rxjs';
import { UsuarioListaComponent } from './usuario-lista.component';
import { UsuarioService } from '../../servicios/usuario.service';

describe('UsuarioListaComponent', () => {
  let fixture: ComponentFixture<UsuarioListaComponent>;
  let component: UsuarioListaComponent;
  let serviceSpy: jasmine.SpyObj<UsuarioService>;

  beforeEach(async () => {
    const spy = jasmine.createSpyObj('UsuarioService', ['getAll', 'delete']);

    await TestBed.configureTestingModule({
      imports: [UsuarioListaComponent],  // standalone
      providers: [{ provide: UsuarioService, useValue: spy }]
    }).compileComponents();

    serviceSpy = TestBed.inject(UsuarioService) as jasmine.SpyObj<UsuarioService>;
    fixture = TestBed.createComponent(UsuarioListaComponent);
    component = fixture.componentInstance;
  });

  it('debe mostrar la lista de usuarios', () => {
    const mockUsuarios = [{ id: 1, nombre: 'Ana', email: 'ana@test.com' }];
    serviceSpy.getAll.and.returnValue(of({ content: mockUsuarios, totalElements: 1 } as any));

    fixture.detectChanges();  // dispara ngOnInit

    const filas = fixture.debugElement.queryAll(By.css('tbody tr'));
    expect(filas.length).toBe(1);

    const nombre = filas[0].query(By.css('.nombre')).nativeElement.textContent;
    expect(nombre.trim()).toBe('Ana');
  });

  it('debe llamar a delete cuando se hace clic en eliminar', () => {
    serviceSpy.delete.and.returnValue(of(undefined));
    spyOn(window, 'confirm').and.returnValue(true);

    component.eliminar(1);

    expect(serviceSpy.delete).toHaveBeenCalledWith(1);
  });
});
