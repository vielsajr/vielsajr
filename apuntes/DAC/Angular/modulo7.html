7.1 Configuración de rutas

app.routes.ts
// app.routes.ts — Definición de rutas
import { Routes } from '@angular/router';
import { authGuard } from './guards/auth.guard';

export const routes: Routes = [
  // Ruta simple
  { path: '', redirectTo: 'inicio', pathMatch: 'full' },
  { path: 'inicio', component: InicioComponent },

  // Rutas protegidas con guard
  { path: 'dashboard',
    component: DashboardComponent,
    canActivate: [authGuard],
    children: [
      { path: '', redirectTo: 'resumen', pathMatch: 'full' },
      { path: 'resumen', component: ResumenComponent },
      { path: 'perfil', component: PerfilComponent },
    ]
  },

  // Lazy loading — el módulo se carga solo cuando se navega
  { path: 'usuarios',
    canActivate: [authGuard],
    loadChildren: () => import('./usuarios/usuarios.routes')
      .then(m => m.usuariosRoutes)
  },

  // Ruta con parámetro
  { path: 'productos/:id', component: ProductoDetalleComponent },

  // Ruta con múltiples parámetros
  { path: 'buscar/:categoria', component: BuscarComponent },

  // Ruta comodín — SIEMPRE al final
  { path: '**', component: NotFoundComponent },
];

7.2 Navegación y lectura de parámetros

navegacion.ts
// En la plantilla:
// <a routerLink="/inicio">Inicio</a>
// <a [routerLink]="['/usuarios', usuario.id]">Ver usuario</a>
// <a routerLink="/buscar" [queryParams]="{q: 'angular', page: 1}">Buscar</a>
// <a routerLinkActive="activo">Activo cuando la ruta está activa</a>
// <router-outlet></router-outlet>  ← donde se renderizan las rutas hijas

// ── Navegación programática ───────────────────────────────────
@Component({...})
export class MiComponente {
  private router = inject(Router);
  private route = inject(ActivatedRoute);

  navegar(): void {
    this.router.navigate(['/usuarios', 42]);
    this.router.navigate(['/buscar'], { queryParams: { q: 'test' } });
    this.router.navigate(['../lista'], { relativeTo: this.route });
    this.router.navigateByUrl('/dashboard');
  }

  // ── Leer parámetros de ruta ───────────────────────────────
  ngOnInit(): void {
    // Parámetro de ruta :id — una sola vez
    const id = this.route.snapshot.paramMap.get('id');

    // Parámetro de ruta reactivo (cambia sin recargar)
    this.route.paramMap.pipe(
      map(params => +params.get('id')!),
      switchMap(id => this.service.getById(id))
    ).subscribe(dato => this.dato = dato);

    // Query params (?page=2&sort=nombre)
    this.route.queryParamMap.subscribe(params => {
      this.page = +(params.get('page') || 1);
      this.sort = params.get('sort') || 'nombre';
    });
  }
}

7.3 Guards

auth.guard.ts
// auth.guard.ts — Guard funcional (Angular 15+)
import { inject } from '@angular/core';
import { Router, CanActivateFn } from '@angular/router';
import { AuthService } from '../servicios/auth.service';

export const authGuard: CanActivateFn = (route, state) => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (auth.estaAutenticado()) {
    return true;
  }

  // Guardar la URL intentada para redirigir después del login
  router.navigate(['/login'], { queryParams: { returnUrl: state.url } });
  return false;
};

// roles.guard.ts — Guard por roles
export const rolesGuard = (rolesPermitidos: string[]): CanActivateFn => {
  return () => {
    const auth = inject(AuthService);
    const router = inject(Router);
    if (auth.tieneAlgunRol(rolesPermitidos)) return true;
    router.navigate(['/forbidden']);
    return false;
  };
};

// Uso en las rutas:
// { path: 'admin', canActivate: [authGuard, rolesGuard(['ADMIN'])] }

// canDeactivate — Confirmar antes de salir (formularios sin guardar)
export const cambiosSinGuardarGuard: CanDeactivateFn<FormularioComponent> =
  (componente) => {
    if (componente.formulario.dirty) {
      return confirm('¿Seguro que quieres salir? Perderás los cambios.');
    }
    return true;
  };
